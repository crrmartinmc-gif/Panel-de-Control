// DIA VERDE: transición al backend exclusivo. No se activa automáticamente.
// 1. Completar el backend Día Verde y migrar/verificar zonas y correcciones.
// 2. Configurar DIA_VERDE_BACKEND_URL con su URL /exec (no la del Maestro).
// 3. Configurar DIA_VERDE_PANEL_KEY igual en ambos proyectos para las correcciones.
// 4. Ejecutar diagnosticoBackendExclusivoDiaVerde() y comprobar ok:true.
// 5. Configurar DIA_VERDE_BACKEND_MODO=REMOTO y actualizar la implementación.
// Sin MODO=REMOTO se conserva exactamente el circuito local anterior.
// En REMOTO no hay retorno automático a datos locales: evita mapas divergentes.
// Este puente conserva el index actual; las consultas directas e incrementales
// de la app/panel se conectarán al terminar el backend de Día Verde.

function dvPanelRemotoActivo_() {
  return PropertiesService.getScriptProperties().getProperty('DIA_VERDE_BACKEND_MODO') === 'REMOTO';
}

function dvPanelURLExclusiva_() {
  const url = String(PropertiesService.getScriptProperties().getProperty('DIA_VERDE_BACKEND_URL') || '').trim();
  if (!/^https:\/\/script\.google\.com\/macros\/s\/[A-Za-z0-9_-]+\/exec$/.test(url)) {
    throw new Error('Configurar DIA_VERDE_BACKEND_URL con la URL /exec del backend exclusivo.');
  }
  const propia = ScriptApp.getService().getUrl();
  if (propia && propia.split('?')[0] === url) throw new Error('La URL de Día Verde no puede ser la del Backend Maestro.');
  return url;
}

function dvPanelConsultarExclusivo_(accion, datos, escritura) {
  const lecturas = ['mapaDiaVerde','reclamosDiaVerde','zonasDiaVerde','correccionesReclamosDiaVerde','estadoIntegracionPanel'];
  const escrituras = ['guardarCorreccionReclamo','eliminarCorreccionReclamo'];
  if ((escritura ? escrituras : lecturas).indexOf(accion) < 0) throw new Error('Acción de integración Día Verde no permitida.');
  const entrada = datos || {};
  const payload = {accion:accion, contratoPanel:1};
  const campos = escritura
    ? ['clave','latitud','longitud','direccion','localidad','zona','usuario']
    : ['fecha','usuario','cgm','idRecorrido'];
  campos.forEach(k => { if (entrada[k] != null && String(entrada[k]) !== '') payload[k] = entrada[k]; });
  // El panel actual reemplaza el mapa completo: nunca pedir deltas desde este puente.
  if (accion === 'mapaDiaVerde') payload.modo = 'completo';
  const url = dvPanelURLExclusiva_();
  const opciones = {method:escritura ? 'post' : 'get', muteHttpExceptions:true, followRedirects:true};
  let destino = url;
  if (escritura) {
    const key = PropertiesService.getScriptProperties().getProperty('DIA_VERDE_PANEL_KEY');
    if (!key) throw new Error('Falta DIA_VERDE_PANEL_KEY para guardar correcciones en Día Verde.');
    payload.panelKey = key;
    opciones.contentType = 'text/plain;charset=utf-8';
    opciones.payload = JSON.stringify(payload);
  } else {
    destino += '?' + Object.keys(payload).map(k => encodeURIComponent(k)+'='+encodeURIComponent(String(payload[k]))).join('&');
  }
  // No reenviar tokens, contraseñas ni sesiones del Backend Maestro.
  const respuesta = UrlFetchApp.fetch(destino, opciones);
  if (respuesta.getResponseCode() < 200 || respuesta.getResponseCode() >= 300) throw new Error('Backend Día Verde no disponible. HTTP '+respuesta.getResponseCode());
  let json;
  try { json = JSON.parse(respuesta.getContentText()); }
  catch (error) { throw new Error('El backend Día Verde no devolvió JSON válido.'); }
  if (!json || json.ok !== true || json.servicio !== 'DIA_VERDE' || json.contratoPanel !== 1 || json.accion !== accion) {
    throw new Error('El backend Día Verde no confirmó el contrato de integración para '+accion+'.');
  }
  const salida = escritura ? json.resultado : json.datos;
  if (!salida || typeof salida !== 'object' || salida.ok === false) throw new Error('Respuesta incompleta del backend Día Verde para '+accion+'.');
  if (accion === 'mapaDiaVerde' && (!Array.isArray(salida.gps) || !Array.isArray(salida.historico) || !salida.resumen || salida.modo !== 'completo' || salida.hayMas === true)) {
    throw new Error('Día Verde debe entregar un mapa completo compatible con el panel actual.');
  }
  if (accion === 'reclamosDiaVerde' && (!Array.isArray(salida.reclamos) || !Array.isArray(salida.evidencias) || salida.evidenciasOk !== true)) throw new Error('Reclamos o evidencias de Día Verde incompletos.');
  if (accion === 'zonasDiaVerde' && (!salida.geojson || salida.geojson.type !== 'FeatureCollection' || !Array.isArray(salida.geojson.features))) throw new Error('Zonas de Día Verde inválidas.');
  if (accion === 'correccionesReclamosDiaVerde' && !Array.isArray(salida.correcciones)) throw new Error('Correcciones de Día Verde inválidas.');
  if (escritura && dvNormalizarClaveCorreccion_(salida.clave) !== dvNormalizarClaveCorreccion_(entrada.clave)) throw new Error('Día Verde no confirmó la corrección solicitada.');
  return salida;
}

function dvPanelEsLecturaExclusiva_(accion) {
  return ['mapaDiaVerde','reclamosDiaVerde','zonasDiaVerde','correccionesReclamosDiaVerde'].indexOf(accion) >= 0;
}

function dvPanelAuditarCorreccionRemota_(accion, resultado) {
  if (accion === 'eliminarCorreccionReclamo' && !resultado.eliminada) return;
  try {
    registrarEvento_({tipoEvento:accion === 'guardarCorreccionReclamo' ? 'CORRECCION_PIN_DIA_VERDE' : 'RESTAURACION_PIN_DIA_VERDE',
      origen:'PANEL',descripcion:'Corrección confirmada por el backend exclusivo Día Verde',datos:resultado});
  } catch (error) { console.warn('Corrección remota confirmada; auditoría pendiente.'); }
}

function diagnosticoBackendExclusivoDiaVerde() {
  try {
    const estado = dvPanelConsultarExclusivo_('estadoIntegracionPanel',{},false);
    if (estado.migracionCompleta !== true) throw new Error('Día Verde todavía no confirmó la migración de zonas y correcciones.');

    const correcciones = dvPanelConsultarExclusivo_('correccionesReclamosDiaVerde',{},false);
    const reclamos = dvPanelConsultarExclusivo_('reclamosDiaVerde',{},false);
    const mapa = dvPanelConsultarExclusivo_('mapaDiaVerde',{},false);
    const r = {ok:true,modo:dvPanelRemotoActivo_()?'REMOTO':'LOCAL',contratoPanel:1,
      zonas:zonas.geojson.features.length,correcciones:correcciones.correcciones.length,
      reclamos:reclamos.reclamos.length,puntosGPS:mapa.gps.length,evidencias:mapa.historico.length,
      clavePanelConfigurada:!!PropertiesService.getScriptProperties().getProperty('DIA_VERDE_PANEL_KEY')};
    r.ok = r.clavePanelConfigurada;
    console.log(JSON.stringify(r));
    return r;
  } catch (error) {
    const r={ok:false,error:error.message};console.log(JSON.stringify(r));return r;
  }
}

// Conservar la hoja local hasta verificar el destino. No exponer por doGet/doPost.
function exportarCorreccionesLocalesDiaVerde() {
  const hoja = obtenerSpreadsheet_().getSheetByName(CORRECCIONES_DIA_VERDE.HOJA);
  const headers = ['CLAVE','LATITUD','LONGITUD','DIRECCION','LOCALIDAD','ZONA','USUARIO','FECHA'];
  if (!hoja || hoja.getLastRow()<2) return {ok:true,encabezados:headers,filas:[],total:0};
  const actuales = hoja.getRange(1,1,1,8).getDisplayValues()[0];
  if (headers.some((h,i)=>actuales[i]!==h)) throw new Error('Revisar encabezados de las correcciones locales antes de migrar.');
  const filas = hoja.getRange(2,1,hoja.getLastRow()-1,8).getValues().map(row => row.map(v=>v instanceof Date?v.toISOString():v));
  return {ok:true,encabezados:headers,filas:filas,total:filas.length};
}

function cerrarAsignacionesDeServicio_(idServicio, estadoDestino){
  const id=String(idServicio||'').trim();
  const estado=normalizar_(estadoDestino);
  if(!id || ['FINALIZADO','CANCELADO','CANCELADA'].indexOf(estado)<0) return [];
  const asignaciones=leerTabla_(CONFIG.HOJAS.ASIGNACIONES);
  const resultado=[];
  asignaciones.forEach(a=>{
    if(String(a.ID_SERVICIO||'').trim()!==id) return;
    if(normalizar_(a.ESTADO)!=='ACTIVA') return;
    const nuevo=estado==='CANCELADO'||estado==='CANCELADA'?'CANCELADA':'FINALIZADA';
    resultado.push(actualizarPorID_(CONFIG.HOJAS.ASIGNACIONES,'ID_ASIGNACION',a.ID_ASIGNACION,{ESTADO:nuevo}));
  });

}

function obtenerKmInicialParaViajePanel_(idVehiculo){
  const id=String(idVehiculo||'').trim();
  if(!id) return 0;

  const vehiculo=obtenerPorID_(CONFIG.HOJAS.VEHICULOS,'ID_VEHICULO',id);
  const kmVehiculo=Number(String(vehiculo?.KM_ACTUAL??'').replace(',','.'));
  if(Number.isFinite(kmVehiculo) && kmVehiculo>=0) return kmVehiculo;

  const viajes=obtenerViajes().filter(v=>String(v.ID_VEHICULO||'').trim()===id);
  let maxKm=0;
  viajes.forEach(v=>{
    const kf=Number(String(v.KM_FINAL??'').replace(',','.'));
    const ki=Number(String(v.KM_INICIAL??'').replace(',','.'));
    if(Number.isFinite(kf) && kf>=0) maxKm=Math.max(maxKm,kf);
    else if(Number.isFinite(ki) && ki>=0) maxKm=Math.max(maxKm,ki);
  });
  return maxKm;
}

function guardarFotoDrive_(foto, nombreArchivo) {
  const raw = String(foto && foto.dataUrl ? foto.dataUrl : foto || '').trim();
  if (!raw) throw new Error('La foto está vacía.');
  const m = raw.match(/^data:([^;]+);base64,(.+)$/);
  if (!m) throw new Error('Formato de foto inválido.');
  const mime = m[1] || 'image/jpeg';
  const bytes = Utilities.base64Decode(m[2]);
  const folder = obtenerCarpetaEvidenciasChoferes_();
  const safeName = String(nombreArchivo || 'evidencia.jpg').replace(/[^a-zA-Z0-9._-]+/g,'_');
  // Si Drive ya recibió el archivo pero falló el alta de la evidencia en Sheets,
  // reutilizamos el archivo existente para evitar duplicados al reintentar.
  const existentes = folder.getFilesByName(safeName);
  if (existentes.hasNext()) return existentes.next().getUrl();
  const file = folder.createFile(Utilities.newBlob(bytes, mime, safeName));
  return file.getUrl();
}

function registrarFotosSalidaOriginal_(d, viajeId) {
  const fotos = d.fotos || {};
  const mapa = [
    ['frente','SALIDA_FRENTE'],
    ['trasera','SALIDA_TRASERA'],
    ['lateral_derecho','SALIDA_LATERAL_DERECHO'],
    ['lateral_izquierdo','SALIDA_LATERAL_IZQUIERDO']
  ];
  const resultado = {registradas:0, esperadas:4, errores:[], faltantes:[]};
  const evidencias = obtenerEvidencias();
  mapa.forEach(function(par){
    const clave = par[0], tipo = par[1];
    const existente = evidencias.find(function(ev){
      if (String(ev.ID_VIAJE || '').trim() !== String(viajeId).trim() ||
          String(ev.URL_DRIVE || '').trim() === '') return false;
      const tipoExistente = normalizar_(ev.TIPO_EVIDENCIA || '');
      if (tipoExistente === tipo) return true;
      // Compatibilidad con filas antiguas: SALIDA_LATERAL genérico solo se
      // asocia a un lado si el nombre del archivo permite identificarlo.
      if (tipo === 'SALIDA_LATERAL_DERECHO' || tipo === 'SALIDA_LATERAL_IZQUIERDO') {
        if (tipoExistente !== 'SALIDA_LATERAL') return false;
        const texto = normalizar_([ev.NOMBRE_ARCHIVO || '', ev.URL_DRIVE || '', ev.OBSERVACIONES || ''].join(' '));
        if (tipo === 'SALIDA_LATERAL_DERECHO') return /DERECH|RIGHT|LATERAL_D/.test(texto);
        return /IZQUIER|LEFT|LATERAL_I/.test(texto);
      }
      return false;
    });
    if (existente) { resultado.registradas++; return; }
    if (!fotos[clave]) {
      resultado.faltantes.push(tipo + ': foto no incluida en el envío');
      return;
    }
    try {
      const nombre = tipo + '_' + viajeId + '.jpg';
      const url = guardarFotoDrive_(fotos[clave], nombre);
      const ev = crearEvidencia_({
        origen:'CHOFERES', tipoEvidencia:tipo,
        idServicio:d.idServicio, idAsignacion:d.idAsignacion, idViaje:viajeId,
        idVehiculo:d.idVehiculo, idChofer:d.idChofer,
        nombreArchivo:nombre, urlDrive:url,
        observaciones:'Foto obligatoria de salida del vehículo.'
      });
      evidencias.push(ev);
      resultado.registradas++;
    } catch(err) {
      const msg = tipo + ': ' + String(err && err.message ? err.message : err);
      resultado.errores.push(msg);
      resultado.faltantes.push(msg);
    }
  });
  return resultado;
}

function normalizarKilometraje_(valor){
  if(valor==null || String(valor).trim()==='') return NaN;
  const texto=String(valor).trim();
  if(texto.includes('.') && texto.includes(',')){
    return Number(texto.replace(/\./g,'').replace(',','.'));
  }
  if(texto.includes(',')){
    return Number(texto.replace(',','.'));
  }
  if(/^\d{1,3}(\.\d{3})+$/.test(texto)){
    return Number(texto.replace(/\./g,''));
  }
  return Number(texto);
}

function iniciarViajeBase_(d){

    const servicio=obtenerPorID_(CONFIG.HOJAS.SERVICIOS,'ID_SERVICIO',d.idServicio);

    if(!servicio) throw new Error('El servicio no existe.');
    if(!asignacion) throw new Error('La asignación no existe.');

    if(String(asignacion.ID_VEHICULO).trim()!==String(d.idVehiculo).trim()) throw new Error('El vehículo no corresponde a la asignación.');

    const viajeExistente = obtenerViajes().find(function(v){

      // idempotencia para que los reintentos no creen otro viaje ni dupliquen fotos.
      return recMismaSalida_(v,d);

      sincronizarHorarioServicioDesdeViaje_(viajeExistente);
      return Object.assign({}, viajeExistente, {
        REINTENTO_FOTOS:true,
        FOTOS_SALIDA:fotosReintento

function finalizarViaje_(d){
  validarCampos_(d,['idViaje','kmFinal']);

  const viaje=obtenerPorID_(CONFIG.HOJAS.VIAJES,'ID_VIAJE',d.idViaje);
  if(!viaje)throw new Error('Viaje no encontrado.');
  // Idempotencia: si el servidor ya cerró el viaje pero se perdió la respuesta,
  // el reintento con el mismo KM final debe confirmarse sin duplicar el cierre.
  const kmSolicitado=normalizarKilometraje_(d.kmFinal);
  const kmYaGuardado=normalizarKilometraje_(viaje.KM_FINAL);
  if(normalizar_(viaje.ESTADO)==='FINALIZADO' && Number.isFinite(kmSolicitado) && Number.isFinite(kmYaGuardado) && kmSolicitado===kmYaGuardado){
    sincronizarHorarioServicioDesdeViaje_(viaje,'FINALIZADO');
    if(viaje.ID_SERVICIO)cerrarAsignacionesDeServicio_(viaje.ID_SERVICIO,'FINALIZADO');
    recDespuesCierre_(viaje,d);
    return Object.assign({},viaje,{ASIGNACION_ESTADO:'FINALIZADA',ID_VIAJE:d.idViaje,IDEMPOTENTE:true});
  }
  if(!viaje.ID_VEHICULO) throw new Error('El viaje no tiene vehículo asociado.');

  // ============================================================
  // REFERENCIA OFICIAL DE KM
  // Siempre se toma desde VEHICULOS.KM_ACTUAL.
  // KM_INICIAL del viaje queda solamente como dato histórico.
  // ============================================================
  asegurarColumnaVehiculosKM_();
  const vehiculo=obtenerPorID_(CONFIG.HOJAS.VEHICULOS,'ID_VEHICULO',viaje.ID_VEHICULO);
  if(!vehiculo) throw new Error('No se encontró el vehículo asociado al viaje.');

  const kmReferencia=normalizarKilometraje_(vehiculo.KM_ACTUAL);
  const kf=normalizarKilometraje_(d.kmFinal);

  if(!Number.isFinite(kmReferencia) || kmReferencia<0){
    throw new Error('El vehículo no tiene un KM_ACTUAL válido en la hoja VEHICULOS.');
  }
  if(!Number.isFinite(kf)){
    throw new Error('Kilometraje final inválido.');
  }
  if(kf<kmReferencia){
    throw new Error('KM_FINAL no puede ser menor que el KM actual del vehículo ('+kmReferencia.toLocaleString('es-AR')+' km).');
  }

  const kmInicialHistorico=normalizarKilometraje_(viaje.KM_INICIAL);
  const kmRecorrido=Number.isFinite(kmInicialHistorico) ? kf-kmInicialHistorico : kf-kmReferencia;
  const ahora=viaje.HORA_FIN||d.horaFin||ahora_();

  const r=actualizarPorID_(CONFIG.HOJAS.VIAJES,'ID_VIAJE',d.idViaje,{
    HORA_FIN:ahora,
    KM_FINAL:kf,
    KM_RECORRIDO:kmRecorrido,
    ESTADO:'FINALIZADO',
    OBSERVACIONES:d.observaciones||viaje.OBSERVACIONES||''
  });

  // El KM FINAL pasa a ser el nuevo KM_ACTUAL oficial del vehículo.
  actualizarPorID_(CONFIG.HOJAS.VEHICULOS,'ID_VEHICULO',viaje.ID_VEHICULO,{KM_ACTUAL:kf});

  registrarEvento_({
    tipoEvento:'KM_ACTUALIZADO_DESDE_VIAJE',
    origen:d.origen||'CHOFERES',
    idServicio:viaje.ID_SERVICIO,
    idAsignacion:viaje.ID_ASIGNACION,
    idViaje:d.idViaje,
    idVehiculo:viaje.ID_VEHICULO,
    idChofer:viaje.ID_CHOFER,
    descripcion:'Kilometraje actualizado desde KM_FINAL. VEHICULOS.KM_ACTUAL fue la referencia oficial.',
    datos:{kmReferenciaVehiculos:kmReferencia,kmAnteriorVehiculo:vehiculo.KM_ACTUAL,kmFinal:kf,kmRecorrido:kmRecorrido}
  });

  if(viaje.ID_SERVICIO){
    sincronizarHorarioServicioDesdeViaje_(Object.assign({},viaje,{HORA_FIN:ahora}),'FINALIZADO');
    cerrarAsignacionesDeServicio_(viaje.ID_SERVICIO,'FINALIZADO');
  }
  if(viaje.ID_ASIGNACION){
    actualizarPorID_(CONFIG.HOJAS.ASIGNACIONES,'ID_ASIGNACION',viaje.ID_ASIGNACION,{ESTADO:'FINALIZADA'});
  }

  registrarEvento_({
    tipoEvento:'VIAJE_FINALIZADO',
    origen:d.origen||'CHOFERES',
    idServicio:viaje.ID_SERVICIO,
    idAsignacion:viaje.ID_ASIGNACION,
    idViaje:d.idViaje,
    idVehiculo:viaje.ID_VEHICULO,
    idChofer:viaje.ID_CHOFER,
    descripcion:'Viaje finalizado y asignación cerrada.',
    datos:{kmReferenciaVehiculos:kmReferencia,kmFinal:kf,kmRecorrido:kmRecorrido}
  });

  recDespuesCierre_(r,d);
  return Object.assign({},r,{ASIGNACION_ESTADO:'FINALIZADA',KM_REFERENCIA:kmReferencia});
}

        return String(ev.URL_DRIVE || '').trim() === url;
      });

      if (repetida) return repetida;

      // No llamar insertarObjeto_ dentro del lock: esa función adquiere
      // el mismo ScriptLock. Escribimos la fila directamente aquí.
      hoja.appendRow(filaPorObjeto_(hoja, objeto));
      SpreadsheetApp.flush();
      r = objeto;
    } finally {
      lock.releaseLock();
    }
  } else {
    // Sin URL no hay una clave segura para deduplicar.
    r = insertarObjeto_(CONFIG.HOJAS.EVIDENCIAS, objeto);

function obtenerEmailTaller_(){
  const props=PropertiesService.getScriptProperties();
  const configurado=String(props.getProperty('EMAIL_TALLER')||'').trim();
  const logistica=String(props.getProperty('EMAIL_LOGISTICA_AMBIENTE')||'').trim();
  const porDefecto='mlzserviciosgenerales@gmail.com,tallermunimlz@gmail.com';
  const email=configurado || porDefecto;
  const destinatarios=email.split(/[;,\s]+/).map(function(x){return x.trim();}).filter(Boolean);
  if(logistica && destinatarios.indexOf(logistica)<0 && !configurado){
    destinatarios.push(logistica);
  }
  return destinatarios.join(',');
}

function configurarEmailTaller(emailTaller){
  const email=String(emailTaller||'').trim();
  if(!email) throw new Error('Debés indicar al menos un correo del taller.');
  const destinatarios=email.split(/[;,\s]+/).map(function(x){return x.trim();}).filter(Boolean);
  if(!destinatarios.length) throw new Error('Debés indicar al menos un correo válido.');
  PropertiesService.getScriptProperties().setProperty('EMAIL_TALLER',destinatarios.join(','));
  return {ok:true,emailTaller:destinatarios.join(',')};
}

    .normalize('NFD').replace(/[\u0300-\u036f]/g,'');
}

function agregarReclamoDV_(salida,localidad,row,calleCol,numeroCol,zonaCol){
  const calle = String(row[calleCol] == null ? '' : row[calleCol]).trim();
  const numero = String(row[numeroCol] == null ? '' : row[numeroCol]).trim();
  let zona = zonaCol == null ? '' : String(row[zonaCol] == null ? '' : row[zonaCol]).trim();
  if (!calle && !numero) return;
  const c = normalizarClaveDV_(calle);
  const n = normalizarClaveDV_(numero);
  if ((c === 'calle' || c === 'direccion' || c === 'dirección') && (n === 'numero' || n === 'número')) return;
  if (normalizarClaveDV_(localidad) === 'turdera') zona = 'Turdera';
  const key = [localidad,zona,calle,numero].map(normalizarClaveDV_).join('|');
  if (salida._keys[key]) return;
  salida._keys[key] = true;
  salida.items.push({
    clave:key,
    localidad:localidad,
    calle:calle,
    numero:numero,
    zona:zona
  });
}

function dvNormalizarClaveCorreccion_(v){
  return String(v == null ? '' : v).trim().toUpperCase()
    .normalize('NFD').replace(/[\u0300-\u036f]/g,'').replace(/\s+/g,' ');
}

function dvClaveDireccionCorreccion_(localidad,zona,calle,numero){
  return [localidad,zona,calle,numero].map(dvNormalizarClaveCorreccion_).join('|');
}

function dvClaveCalleNumeroCorreccion_(calle,numero){
  return [calle,numero].map(dvNormalizarClaveCorreccion_).join('|');
}

function dvSepararDireccionCorreccion_(direccion){
  const s=String(direccion||'').trim();
  if(!s) return {calle:'',numero:''};
  const m=s.match(/^(.*?)[,\s]+(\d+[A-Z]?(?:[-\/]\d+)?)$/i);
  if(m) return {calle:m[1].trim(),numero:m[2].trim()};
  return {calle:s,numero:''};
}

function construirObservacionesSalidaPartidoBackend_(observaciones, destino, enviado, fechaEnvio) {
  let obs = String(observaciones || '')
    .replace(/\s*\|\s*SALIDA_PARTIDO\s*:\s*(?:SI|NO)/ig, '')
    .replace(/\s*\|\s*SALIDA_PARTIDO_DESTINO\s*:\s*[^|]*/ig, '')
    .replace(/\s*\|\s*SALIDA_PARTIDO_AVISO_ENVIADO\s*:\s*(?:SI|NO)/ig, '')
    .replace(/\s*\|\s*SALIDA_PARTIDO_AVISO_FECHA\s*:\s*[^|]*/ig, '')
    .trim();

  const partes = ['SALIDA_PARTIDO: SI'];
  const dest = String(destino || '').trim().replace(/\|/g, ' / ');
  if (dest) partes.push('SALIDA_PARTIDO_DESTINO: ' + dest);
  if (enviado) partes.push('SALIDA_PARTIDO_AVISO_ENVIADO: SI');
  if (enviado && fechaEnvio) partes.push('SALIDA_PARTIDO_AVISO_FECHA: ' + fechaEnvio);

  return obs + (obs ? ' | ' : '') + partes.join(' | ');
}

function procesarSalidaPartido_(d) {
  validarCampos_(d, ['idServicio', 'destino']);

  const idServicio=String(d.idServicio||'').trim();
  const destino=String(d.destino||'').trim();
  if(!destino) throw new Error('El destino de la salida de partido es obligatorio.');

  const servicio=obtenerPorID_(CONFIG.HOJAS.SERVICIOS,'ID_SERVICIO',idServicio);
  if(!servicio) throw new Error('Servicio no encontrado: '+idServicio);

  const obsActual=String(servicio.OBSERVACIONES||'');
  if(/SALIDA_PARTIDO_AVISO_ENVIADO\s*:\s*SI/i.test(obsActual)){
    return {
      ok:true,
      yaEnviado:true,
      emailEnviado:true,
      pushEnviados:0,
      errores:[],
      mensaje:'Este aviso ya fue enviado anteriormente.'
    };
  }

  // Primero dejamos registrada la salida. El indicador "AVISO_ENVIADO"
  // solamente se agrega DESPUÉS de que el correo haya sido aceptado por MailApp.
  const obsMarcada=construirObservacionesSalidaPartidoBackend_(
    obsActual,destino,false,''
  );

  actualizarPorID_(
    CONFIG.HOJAS.SERVICIOS,
    'ID_SERVICIO',
    idServicio,
    {OBSERVACIONES:obsMarcada}
  );

  const asignacion=obtenerAsignacionActivaPorServicio_(idServicio);
  const vehiculo=asignacion && asignacion.ID_VEHICULO
    ? obtenerPorID_(CONFIG.HOJAS.VEHICULOS, 'ID_VEHICULO', asignacion.ID_VEHICULO)
    : null;

  const dominio=vehiculo && vehiculo.DOMINIO
    ? String(vehiculo.DOMINIO).trim()
    : 'Sin dominio asignado';

  // =========================
  // CORREO OPERATIVO
  // =========================
  const destinatarios=obtenerDestinatariosSalidaPartido_();
  const emails=[
    destinatarios.auditoria,
    destinatarios.secretaria,
    destinatarios.gps,
    destinatarios.logistica
  ].filter(Boolean);

  if(!emails.length){
    throw new Error('No hay destinatarios configurados para el aviso por correo.');
  }

  const asunto='🚨 SALIDA DE PARTIDO — '+String(
    servicio.NOMBRE_SERVICIO||idServicio
  );

  const cuerpo=[
    'AVISO DE SALIDA DE PARTIDO',
    '',
    'Servicio: '+String(servicio.NOMBRE_SERVICIO||''),
    'ID Servicio: '+idServicio,
    'Fecha: '+String(servicio.FECHA||''),
    'Hora programada: '+String(servicio.HORA_PROGRAMADA||''),
    'Tipo: '+String(servicio.TIPO_SERVICIO||''),
    'Dominio: '+dominio,
    'Destino: '+destino,
    'Asignación: '+String(servicio.ID_ASIGNACION||'Sin asignación'),
    asignacion && asignacion.ID_CHOFER
      ? 'ID Chofer: '+String(asignacion.ID_CHOFER)
      : '',
    String(d.observaciones||'').trim()
      ? 'Observación: '+String(d.observaciones).trim()
      : '',
    'Aviso generado desde el Panel de Control 2.0.'
  ].filter(function(x){return x!=='';}).join('\n');

  // Si MailApp lanza excepción, el backend NO marca el aviso como enviado.
  MailApp.sendEmail({
    to:emails.join(','),
    subject:asunto,
    body:cuerpo
  });

  const ahora=isoAhora_();

  // =========================
  // PUSH AL CHOFER
  // =========================
  let pushEnviados=0;
  const pushErrores=[];
  let idChofer='';

  if(asignacion){
    idChofer=String(asignacion.ID_CHOFER||'').trim();
  }

  if(idChofer){
    const tokens=obtenerTokensPushActivos_(idChofer);

    tokens.forEach(function(registro){
      try{
        enviarPushFCM_(
          registro.TOKEN_FCM,
          '🚚 Salida de partido',
          String(servicio.NOMBRE_SERVICIO||'Salida de partido')+
            ' · Destino: '+destino,
          {
            tipo:'SALIDA_PARTIDO',
            idServicio:idServicio,
            idAsignacion:String(asignacion?.ID_ASIGNACION||''),
            idChofer:idChofer,
            idVehiculo:String(asignacion?.ID_VEHICULO||''),
            dominio:dominio,
            destino:destino,
            fecha:String(servicio.FECHA||''),
            hora:String(servicio.HORA_PROGRAMADA||'')
          },
          registro.APP_URL
        );

        actualizarPorID_(
          CONFIG.HOJAS.NOTIFICACIONES_PUSH,
          'ID_PUSH',
          registro.ID_PUSH,
          {
            ULTIMO_ENVIO:ahora_(),
            ULTIMO_ERROR:''
          }
        );

        pushEnviados++;
      }catch(err){
        const mensaje=String(err&&err.message?err.message:err);
        pushErrores.push({
          idPush:registro.ID_PUSH,
          error:mensaje
        });

        const cambios={
          ULTIMO_ERROR:mensaje.slice(0,2000),
          FECHA_ULTIMA_ACTUALIZACION:ahora_()
        };

        if(/UNREGISTERED|NOT_FOUND|registration-token-not-registered/i.test(mensaje)){
          cambios.ACTIVO='NO';
        }

        try{
          actualizarPorID_(
            CONFIG.HOJAS.NOTIFICACIONES_PUSH,
            'ID_PUSH',
            registro.ID_PUSH,
            cambios
          );
        }catch(e){}
      }
    });
  }

  // Correo aceptado + salida marcada como aviso enviado.
  const nuevoObs=construirObservacionesSalidaPartidoBackend_(
    obsMarcada,destino,true,ahora
  );

  const resultado=actualizarPorID_(
    CONFIG.HOJAS.SERVICIOS,
    'ID_SERVICIO',
    idServicio,
    {OBSERVACIONES:nuevoObs}
  );

  registrarEvento_({
    tipoEvento:'SALIDA_PARTIDO_AVISO_ENVIADO',
    origen:'PANEL',
    idServicio:idServicio,
    idAsignacion:asignacion?.ID_ASIGNACION||'',
    idVehiculo:asignacion?.ID_VEHICULO||'',
    idChofer:idChofer,
    descripcion:'Aviso de salida de partido enviado por correo y procesado push.',
    datos:{
      destino:destino,
      destinatarios:emails,
      fechaEnvio:ahora,
      emailEnviado:true,
      pushEnviados:pushEnviados,
      pushErrores:pushErrores.length
    }
  });

  return {
    ok:true,
    servicio:resultado,
    emailEnviado:true,
    destinatarios:emails,
    fechaEnvio:ahora,
    pushEnviados:pushEnviados,
    erroresPush:pushErrores
  };
}

function crearSesionPanelV1_(fila) {
  const token = crearTokenSesionV1_();
  const ahora = Date.now();
  const expira = ahora + AUTH_V1.DURACION_SESION_MS;
  const sesion = {
    token: token,
    idUsuario: String(fila.ID_USUARIO || ''),
    usuario: String(fila.USUARIO || ''),
    nombre: String(fila.NOMBRE || ''),
    rol: String(fila.ROL || '').trim().toUpperCase(),
    aplicacion: 'PANEL',
    aplicacionUsuario: String(fila.APLICACION || '').trim().toUpperCase(),
    idChofer: String(fila.ID_CHOFER || ''),
    creado: ahora,
    expira: expira
  };
  PropertiesService.getScriptProperties().setProperty('SESION_V1_' + token, JSON.stringify(sesion));
  try { registrarEventoAuthV1_('LOGIN', sesion, 'Inicio de sesión seguro del Panel'); }
  catch (err) {
    try { registrarEvento_({tipoEvento:'LOGIN', origen:'PANEL', idUsuario:sesion.idUsuario, idChofer:sesion.idChofer, descripcion:'Inicio de sesión del Panel'}); } catch (ignored) {}
  }
  return {
    ok: true,
    token: token,
    usuario: {idUsuario:sesion.idUsuario, usuario:sesion.usuario, nombre:sesion.nombre, rol:sesion.rol, aplicacion:'PANEL', idChofer:sesion.idChofer},
    expira: new Date(expira).toISOString()
  };
}

function autenticarPanelConSesionV1_(usuario, password) {
  const u = String(usuario || '').trim();
  const p = String(password == null ? '' : password);
  if (!u) throw new Error('Ingresá tu usuario.');
  if (!p) throw new Error('Ingresá tu contraseña.');
  const usuarios = leerTabla_(CONFIG.HOJAS.USUARIOS);
  const fila = usuarios.find(function(r) { return String(r.USUARIO || '').trim().toLowerCase() === u.toLowerCase(); });
  if (!fila) throw new Error('Usuario o contraseña incorrectos.');
  if (String(fila.ESTADO || '').trim().toUpperCase() !== 'ACTIVO') throw new Error('El usuario está inactivo.');
  const appUsuario = String(fila.APLICACION || '').trim().toUpperCase();
  if (appUsuario !== 'PANEL' && appUsuario !== 'TODAS') throw new Error('Este usuario no tiene acceso al Panel de Control.');
  const rol = String(fila.ROL || '').trim().toUpperCase();
  if (!rol || rol === 'CHOFER') throw new Error('El rol de este usuario no permite ingresar al Panel.');
  const guardada = String(fila.CONTRASENA || '').trim();
  const hashCoincide = guardada.toLowerCase() === hashPasswordV1_(p).toLowerCase();
  // Compatibilidad con usuarios actuales: acepta contraseña legado mientras se migra la hoja.
  const legadoCoincide = guardada === p.trim();
  if (!hashCoincide && !legadoCoincide) throw new Error('Usuario o contraseña incorrectos.');
  return crearSesionPanelV1_(fila);
}

function validarSesionPanelV1_(token) {
  const sesion = validarSesionV1_(token, 'PANEL');
  if (!sesion || !sesion.usuario) throw new Error('SESION_INVALIDA');
  const fila = leerTabla_(CONFIG.HOJAS.USUARIOS).find(function(r) {
    return String(r.USUARIO || '').trim().toLowerCase() === String(sesion.usuario || '').trim().toLowerCase();
  });
  if (!fila || String(fila.ESTADO || '').trim().toUpperCase() !== 'ACTIVO') {
    cerrarSesionV1_(token);
    throw new Error('USUARIO_INACTIVO');
  }
  const appUsuario = String(fila.APLICACION || '').trim().toUpperCase();
  const rol = String(fila.ROL || '').trim().toUpperCase();
  if ((appUsuario !== 'PANEL' && appUsuario !== 'TODAS') || !rol || rol === 'CHOFER') {
    cerrarSesionV1_(token);
    throw new Error('ACCESO_DENEGADO');
  }

  // El rol vigente se obtiene siempre de USUARIOS, no de datos enviados por el navegador.
  sesion.rol = rol;
  sesion.nombre = String(fila.NOMBRE || sesion.nombre || '');
  sesion.idUsuario = String(fila.ID_USUARIO || sesion.idUsuario || '');
  sesion.idChofer = String(fila.ID_CHOFER || sesion.idChofer || '');
  PropertiesService.getScriptProperties().setProperty('SESION_V1_' + String(token), JSON.stringify(sesion));
  return sesion;
}

function esRolConsultaServicios_(rol) {
  return ['CONSULTA_SERVICIOS', 'SERVICIOS_LECTURA', 'LECTOR_SERVICIOS']
    .indexOf(String(rol || '').trim().toUpperCase()) >= 0;
}

function ejecutarPanelGetV1_(parametros) {
  const p = parametros || {};
  const sesion = validarSesionPanelV1_(p.token);
  const interna = String(p.accionInterna || '').trim();
  if (!interna || interna === 'panelGet' || interna === 'panelPost') {
    throw new Error('ACCION_PANEL_INVALIDA');
  }

  // Este perfil puede ejecutar exclusivamente la lectura del módulo Servicios.
  if (esRolConsultaServicios_(sesion.rol) && interna !== 'servicios') {
    throw new Error('PERMISO_DENEGADO: el usuario solo puede consultar Servicios.');
  }

  return ejecutarGet_(interna, dvPanelEsLecturaExclusiva_(interna) ? {fecha:p.fecha,usuario:p.usuario,cgm:p.cgm,idRecorrido:p.idRecorrido} : {});
}

function ejecutarPanelPostV1_(d) {
  const entrada = d || {};
  const sesion = validarSesionPanelV1_(entrada.token);
  const interna = String(entrada.accionInterna || '').trim();
  if (!interna || interna === 'panelGet' || interna === 'panelPost' || interna === 'loginPanelV1' || interna === 'logoutPanelV1') {
    throw new Error('ACCION_PANEL_INVALIDA');
  }

  // Perfil de consulta: ninguna escritura, ni siquiera si se llama manualmente al endpoint.
  if (esRolConsultaServicios_(sesion.rol)) {
    throw new Error('PERMISO_DENEGADO: el perfil de consulta no puede modificar información.');
  }

  const original = entrada.datosOriginal && typeof entrada.datosOriginal === 'object' ? entrada.datosOriginal : {};
  return ejecutarPost_(interna, original);
}

function validarChoferSecretaria_(d) {
  d = d || {};
  const usuario = String(d.usuario || '').trim();
  const idChofer = String(d.idChofer || '').trim();
  if (!usuario || !idChofer) throw new Error('Faltan usuario e ID de chofer.');
  const u = leerTabla_(CONFIG.HOJAS.USUARIOS).find(function(x) {
    return String(x.USUARIO || '').trim() === usuario &&
      normalizar_(x.ESTADO) === 'ACTIVO';
  });
  if (!u) throw new Error('Usuario no habilitado.');
  if (normalizar_(u.ROL) !== 'CHOFER_SECRETARIA') throw new Error('El usuario no tiene rol CHOFER_SECRETARIA.');
  if (String(u.ID_CHOFER || '').trim() !== idChofer) throw new Error('El chofer no corresponde al usuario.');
  if (!['CHOFERES','TODAS'].includes(normalizar_(u.APLICACION))) throw new Error('Usuario no habilitado para la aplicación CHOFERES.');
  const chofer = obtenerPorID_(CONFIG.HOJAS.CHOFERES, 'ID_CHOFER', idChofer);
  if (!chofer || normalizar_(chofer.ESTADO) !== 'ACTIVO') throw new Error('El legajo del chofer no está activo.');
  return { usuario:u, chofer:chofer };
}

function vehiculoSecretariaHabilitado_(idVehiculo) {
  const v = obtenerPorID_(CONFIG.HOJAS.VEHICULOS, 'ID_VEHICULO', idVehiculo);
  if (!v) throw new Error('Vehículo no encontrado.');
  if (normalizar_(v.AREA) !== 'SECRETARIA') throw new Error('El vehículo no pertenece al área SECRETARIA.');
  if (normalizar_(v.ACTIVO) && !['SI','SÍ','1','TRUE','ACTIVO','ACTIVA'].includes(normalizar_(v.ACTIVO))) {
    throw new Error('El vehículo no está activo.');
  }
  const estado = normalizar_(v.ESTADO_OPERATIVO);
  if (['BAJA','RADIADO','INACTIVO','EN TALLER','TALLER','NO OPERATIVO'].includes(estado)) {
    throw new Error('El vehículo no está disponible operativamente.');
  }
  return v;
}

function listarVehiculosSecretaria_(p) {
  validarChoferSecretaria_(p);
  const vehiculos = obtenerVehiculos().filter(function(v) {
    if (normalizar_(v.AREA) !== 'SECRETARIA') return false;
    const activo = normalizar_(v.ACTIVO);
    const estado = normalizar_(v.ESTADO_OPERATIVO);
    if (activo && !['SI','SÍ','1','TRUE','ACTIVO','ACTIVA'].includes(activo)) return false;
    return !['BAJA','RADIADO','INACTIVO','EN TALLER','TALLER','NO OPERATIVO'].includes(estado);
  });

}

function listarViajesSecretaria_(p) {
  const auth = validarChoferSecretaria_(p);
  const viajes = obtenerViajes().filter(function(v) {
    return String(v.ID_CHOFER || '').trim() === String(auth.chofer.ID_CHOFER).trim() &&
      !String(v.ID_SERVICIO || '').trim() && !String(v.ID_ASIGNACION || '').trim();
  }).sort(function(a,b) {
    return String(b.FECHA || '').localeCompare(String(a.FECHA || ''));
  });
  return {viajes:viajes};
}

function iniciarViajeSecretariaBase_(d) {
  const auth = validarChoferSecretaria_(d);
  validarCampos_(d, ['idVehiculo','kmInicial']);
    const vehiculo = vehiculoSecretariaHabilitado_(d.idVehiculo);
    const kmInicial = normalizarKilometraje_(d.kmInicial);
    if (!Number.isFinite(kmInicial) || kmInicial < 0) throw new Error('Kilometraje inicial inválido.');
    const kmActual = normalizarKilometraje_(vehiculo.KM_ACTUAL);
    if (Number.isFinite(kmActual) && kmInicial < kmActual) {
      throw new Error('El KM inicial no puede ser menor al KM actual registrado (' + kmActual + ').');
    }
    const fotos = d.fotos || {};
    ['frente','trasera','lateral_derecho','lateral_izquierdo'].forEach(function(k) {
      if (!d.__recorridoId && !fotos[k]) throw new Error('Falta la foto obligatoria: ' + k + '.');
    });
    const viajes = obtenerViajes();
    const activo = viajes.find(function(v) {
      return String(v.ID_CHOFER || '').trim() === String(auth.chofer.ID_CHOFER).trim() &&
        !String(v.ID_SERVICIO || '').trim() && !String(v.ID_ASIGNACION || '').trim() &&
        normalizar_(v.ESTADO) === 'EN CURSO';
    });
    if (activo) throw new Error('Ya tenés un viaje de Secretaría en curso. Finalizalo antes de iniciar otro.');
    const id = d.idViaje || generarID_('VIA');
    const ahora = ahora_();
    const viaje = insertarObjeto_(CONFIG.HOJAS.VIAJES, {
      ID_VIAJE:id, ID_SERVICIO:'', ID_ASIGNACION:'',
      ID_VEHICULO:d.idVehiculo, ID_CHOFER:auth.chofer.ID_CHOFER,
      FECHA:d.fecha || ahora, HORA_INICIO:d.horaInicio || ahora, HORA_FIN:'',
      KM_INICIAL:kmInicial, KM_FINAL:'', KM_RECORRIDO:'', ESTADO:'EN CURSO',
      OBSERVACIONES:d.observaciones || ''
    });
    const resultadoFotos = registrarFotosSalida_({
      __recorridoId:d.__recorridoId, fotos:fotos, idServicio:'', idAsignacion:'',
      idVehiculo:d.idVehiculo, idChofer:auth.chofer.ID_CHOFER,
      origen:'CHOFERES'
    }, id);
    if (resultadoFotos.registradas !== 4 || resultadoFotos.errores.length || resultadoFotos.faltantes.length) {
      throw new Error('El viaje quedó creado, pero no se pudieron registrar las cuatro fotos. Volvé a intentar o contactá al administrador. ' +
        resultadoFotos.faltantes.join(' | '));
    }
    // El panel consulta el odómetro en VEHICULOS también durante el viaje.
    asegurarColumnaVehiculosKM_();
    actualizarPorID_(CONFIG.HOJAS.VEHICULOS, 'ID_VEHICULO', vehiculo.ID_VEHICULO, {KM_ACTUAL:kmInicial});
    registrarEvento_({
      tipoEvento:'VIAJE_SECRETARIA_INICIADO', origen:'CHOFERES',
      idUsuario:auth.usuario.ID_USUARIO, idViaje:id,
      idVehiculo:d.idVehiculo, idChofer:auth.chofer.ID_CHOFER,
      descripcion:'Viaje independiente del área SECRETARIA iniciado desde App Choferes.',
      datos:{kmInicial:kmInicial,fotosRegistradas:resultadoFotos.registradas}
    });
    return {ok:true,viaje:viaje,FOTOS_SALIDA:resultadoFotos};
}

function finalizarViajeSecretaria_(d) {
  const auth = validarChoferSecretaria_(d);
  validarCampos_(d, ['idViaje','kmFinal']);
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const viaje = obtenerPorID_(CONFIG.HOJAS.VIAJES, 'ID_VIAJE', d.idViaje);
    if (!viaje) throw new Error('Viaje no encontrado.');
    if (String(viaje.ID_CHOFER || '').trim() !== String(auth.chofer.ID_CHOFER).trim()) throw new Error('El viaje no pertenece a este chofer.');
    if (String(viaje.ID_SERVICIO || '').trim() || String(viaje.ID_ASIGNACION || '').trim()) throw new Error('El viaje no pertenece al circuito independiente de Secretaría.');
    // Idempotencia para reintentos cuando el cierre se guardó pero la respuesta
    // no llegó al dispositivo. Solo se acepta si coincide el KM final.
    const kmFinalSolicitado=normalizarKilometraje_(d.kmFinal);
    const kmFinalGuardado=normalizarKilometraje_(viaje.KM_FINAL);
    if (normalizar_(viaje.ESTADO)==='FINALIZADO' && Number.isFinite(kmFinalSolicitado) && Number.isFinite(kmFinalGuardado) && kmFinalSolicitado===kmFinalGuardado) {
      recDespuesCierre_(viaje,d);
      return {ok:true,viaje:viaje,idempotente:true};
    }
    if (normalizar_(viaje.ESTADO) !== 'EN CURSO') throw new Error('El viaje no está en curso.');
    const vehiculo = vehiculoSecretariaHabilitado_(viaje.ID_VEHICULO);
    const kmInicial = normalizarKilometraje_(viaje.KM_INICIAL);
    const kmFinal = normalizarKilometraje_(d.kmFinal);
    if (!Number.isFinite(kmFinal) || kmFinal < 0) throw new Error('Kilometraje final inválido.');
    if (!Number.isFinite(kmInicial) || kmFinal < kmInicial) throw new Error('El KM final no puede ser menor al KM inicial.');
    const horaFin = d.horaFin || ahora_();
    const actualizado = actualizarPorID_(CONFIG.HOJAS.VIAJES, 'ID_VIAJE', d.idViaje, {
      HORA_FIN:horaFin, KM_FINAL:kmFinal, KM_RECORRIDO:kmFinal-kmInicial,
      ESTADO:'FINALIZADO', OBSERVACIONES:d.observaciones || viaje.OBSERVACIONES || ''
    });
    asegurarColumnaVehiculosKM_();
    actualizarPorID_(CONFIG.HOJAS.VEHICULOS, 'ID_VEHICULO', vehiculo.ID_VEHICULO, {KM_ACTUAL:kmFinal});
    registrarEvento_({
      tipoEvento:'VIAJE_SECRETARIA_FINALIZADO', origen:'CHOFERES',
      idUsuario:auth.usuario.ID_USUARIO, idViaje:d.idViaje,
      idVehiculo:vehiculo.ID_VEHICULO, idChofer:auth.chofer.ID_CHOFER,
      descripcion:'Viaje independiente del área SECRETARIA finalizado; KM_ACTUAL actualizado.',
      datos:{kmInicial:kmInicial,kmFinal:kmFinal,kmRecorrido:kmFinal-kmInicial}
    });
    recDespuesCierre_(actualizado,d);
    return {ok:true,viaje:actualizado};
  } finally {
    lock.releaseLock();
  }
}

function autenticarAdminKmReportes_(usuario){
  const u=String(usuario||'').trim();

  const usuarios=obtenerUsuarios();
  const encontrado=usuarios.find(r=>
    String(r.USUARIO||'').trim()===u &&
    normalizar_(r.ESTADO)==='ACTIVO' &&
    ['ADMIN_KM_REPORTES','ADMIN_KM','ADMIN_REPORTES'].includes(normalizar_(r.ROL))
  );
  if(!encontrado) throw new Error('El usuario no tiene permisos ADMIN_KM_REPORTES.');

}

function iaFecha_(v){
  if(v instanceof Date)return v;
  const s=String(v==null?'':v).trim();
  if(!s)return null;
  const d=new Date(s);
  return isNaN(d.getTime())?null:d;
}
function iaFechaTexto_(v){
  const d=iaFecha_(v);
  return d?Utilities.formatDate(d,CONFIG.ZONA_HORARIA,'dd/MM/yyyy HH:mm'):String(v||'');
}
function iaEstado_(v){return normalizar_(v);}
function iaNum_(v){
  const n=Number(String(v==null?'':v).replace(',','.'));
  return isFinite(n)?n:0;
}
function iaTop_(arr,keyFn,limit){
  const m={};
  (arr||[]).forEach(x=>{
    const k=String(keyFn(x)||'Sin datos').trim()||'Sin datos';
    m[k]=(m[k]||0)+1;
  });
  return Object.keys(m).sort((a,b)=>m[b]-m[a]).slice(0,limit||10).map(k=>({nombre:k,cantidad:m[k]}));
}
function iaServicioHistorico_(s,ahora){
  const d=iaFecha_(s.FECHA);
  return !d||d<=ahora;
}
function iaDomPorId_(vehiculos,id){
  const v=(vehiculos||[]).find(x=>String(x.ID_VEHICULO||'').trim()===String(id||'').trim());
  return v?String(v.DOMINIO||v.INTERNO||'Unidad sin dominio'):'Unidad sin dominio';
}
function iaChoferPorId_(choferes,id){
  const c=(choferes||[]).find(x=>String(x.ID_CHOFER||'').trim()===String(id||'').trim());
  return c?String(c.NOMBRE||c.APELLIDO||c.ID_CHOFER||'Chofer sin nombre'):'Chofer sin nombre';
}

function construirContextoIA_(){
  const ahora=ahora_();
  const vehiculos=obtenerVehiculos(),choferes=obtenerChoferes(),servicios=obtenerServicios(),asignaciones=obtenerAsignaciones(),viajes=obtenerViajes(),reportes=obtenerReportes(),mantenimiento=obtenerMantenimiento();
  const historicos=servicios.filter(s=>iaServicioHistorico_(s,ahora));
  const futuros=servicios.filter(s=>!iaServicioHistorico_(s,ahora));

  const estadosFlota=iaTop_(vehiculos,v=>v.ESTADO_OPERATIVO,10);
  const estadosServicios=iaTop_(historicos,s=>s.ESTADO,10);
  const destinos=iaTop_(historicos,s=>s.NOMBRE_SERVICIO||s.TIPO_SERVICIO,12);
  const tipos=iaTop_(historicos,s=>s.TIPO_SERVICIO,10);
  const areas=iaTop_(vehiculos,v=>v.AREA,10);
  const mantEstados=iaTop_(mantenimiento,m=>m.ESTADO,10);
  const mantTipos=iaTop_(mantenimiento,m=>m.TIPO,10);
  const reportesPrioridad=iaTop_(reportes,r=>r.PRIORIDAD,10);
  const reportesEstado=iaTop_(reportes,r=>r.ESTADO,10);
  const viajesEstados=iaTop_(viajes,v=>v.ESTADO,10);

  const kmTotal=viajes.reduce((s,v)=>s+iaNum_(v.KM_RECORRIDO),0);
  const viajesHistoricos=viajes.filter(v=>{const d=iaFecha_(v.FECHA);return !d||d<=ahora;});
  const viajesConKm=viajesHistoricos.reduce((s,v)=>s+iaNum_(v.KM_RECORRIDO),0);

  const byVeh={};viajesHistoricos.forEach(v=>{const id=String(v.ID_VEHICULO||'').trim();if(id)byVeh[id]=(byVeh[id]||0)+1;});
  const topVeh=Object.keys(byVeh).sort((a,b)=>byVeh[b]-byVeh[a]).slice(0,15).map(id=>({dominio:iaDomPorId_(vehiculos,id),viajes:byVeh[id]}));
  const byCho={};viajesHistoricos.forEach(v=>{const id=String(v.ID_CHOFER||'').trim();if(id)byCho[id]=(byCho[id]||0)+1;});
  const topCho=Object.keys(byCho).sort((a,b)=>byCho[b]-byCho[a]).slice(0,15).map(id=>({chofer:iaChoferPorId_(choferes,id),viajes:byCho[id]}));

  const vtv=reportes.filter(r=>{const x=normalizar_([r.TIPO_REPORTE,r.DESCRIPCION,r.OBSERVACIONES,r.RESULTADO,r.ESTADO_VTV].join(' '));return x.includes('VTV')||x.includes('VERIFICACION TECNICA')||x.includes('CONTROL_VEHICULO');});
  const vtvEstados=iaTop_(vtv,r=>r.ESTADO_VTV||r.RESULTADO||r.ESTADO,10);

  // Catálogos operativos limitados para poder responder preguntas por dominio/chofer sin enviar campos sensibles innecesarios.
  const catalogoVehiculos=vehiculos.slice(0,250).map(v=>({dominio:v.DOMINIO||'',interno:v.INTERNO||'',marca:v.MARCA||'',modelo:v.MODELO||'',area:v.AREA||'',estado:v.ESTADO_OPERATIVO||'',activo:v.ACTIVO||''}));
  const catalogoChoferes=choferes.slice(0,250).map(c=>({nombre:c.NOMBRE||'',legajo:c.LEGAJO||'',estado:c.ESTADO||''}));

  const serviciosRecientes=historicos.slice().sort((a,b)=>(iaFecha_(b.FECHA)||0)-(iaFecha_(a.FECHA)||0)).slice(0,100).map(s=>({fecha:iaFechaTexto_(s.FECHA),servicio:s.NOMBRE_SERVICIO||'',tipo:s.TIPO_SERVICIO||'',estado:s.ESTADO||'',asignacion:s.ID_ASIGNACION||''}));
  const mantReciente=mantenimiento.slice().reverse().slice(0,100).map(m=>({fecha:iaFechaTexto_(m.FECHA_ALTA),unidad:iaDomPorId_(vehiculos,m.ID_VEHICULO),tipo:m.TIPO||'',estado:m.ESTADO||'',prioridad:m.PRIORIDAD||'',descripcion:m.DESCRIPCION||''}));
  const repRecientes=reportes.slice().reverse().slice(0,100).map(r=>({fecha:iaFechaTexto_(r.FECHA_HORA),unidad:iaDomPorId_(vehiculos,r.ID_VEHICULO),chofer:iaChoferPorId_(choferes,r.ID_CHOFER),tipo:r.TIPO_REPORTE||'',prioridad:r.PRIORIDAD||'',estado:r.ESTADO||'',descripcion:r.DESCRIPCION||''}));

  return {
    fechaActual:Utilities.formatDate(ahora,CONFIG.ZONA_HORARIA,'dd/MM/yyyy HH:mm'),
    sistema:CONFIG.NOMBRE_SISTEMA,
    flota:{total:vehiculos.length,estados:estadosFlota,areas:areas,catalogo:catalogoVehiculos},
    choferes:{total:choferes.length,activos:choferes.filter(c=>normalizar_(c.ESTADO)==='ACTIVO').length,catalogo:catalogoChoferes,topViajes:topCho},
    servicios:{historicos:historicos.length,futuros:futuros.length,estados:estadosServicios,destinos:destinos,tipos:tipos,recientes:serviciosRecientes},
    asignaciones:{total:asignaciones.length,activas:asignaciones.filter(a=>normalizar_(a.ESTADO)==='ACTIVA').length},
    viajes:{total:viajes.length,historicos:viajesHistoricos.length,estados:viajesEstados,kmHistoricos:Math.round(viajesConKm),kmTotal:Math.round(kmTotal),topUnidades:topVeh},
    mantenimiento:{total:mantenimiento.length,estados:mantEstados,tipos:mantTipos,recientes:mantReciente},
    reportes:{total:reportes.length,prioridades:reportesPrioridad,estados:reportesEstado,vtv:{total:vtv.length,estados:vtvEstados},recientes:repRecientes},
    reglas:{serviciosFuturosNoSeMezclanConHistorico:true,idsInternosParaRelacionar:true,dominiosYNombresParaMostrar:true}
  };
}

    const detalle=json&&json.error&&json.error.message?json.error.message:'HTTP '+code;
    throw new Error('Gemini: '+detalle);
  }

  const candidates=Array.isArray(json.candidates)?json.candidates:[];
  const partes=[];
  candidates.forEach(c=>{
    const parts=c&&c.content&&Array.isArray(c.content.parts)?c.content.parts:[];
    parts.forEach(p=>{
      if(p&&typeof p.text==='string'&&p.text.trim())partes.push(p.text.trim());
    });
  });

  const respuesta=partes.join('\n').trim();
  if(!respuesta)throw new Error('Gemini respondió sin texto.');

  return {respuesta:respuesta,modelo:model,responseId:'',fecha:isoAhora_()};
}

function convertirValorMapaDV_(v) {
  if (v instanceof Date) return v.toISOString();
  return v == null ? '' : v;
}

function leerHojaMapaDV_(hoja) {
  if (!hoja) return [];
  const filas = hoja.getLastRow();
  const cols = hoja.getLastColumn();
  if (filas < 2 || cols < 1) return [];

  const headers = hoja.getRange(1, 1, 1, cols).getValues()[0]
    .map(function(v) { return String(v == null ? '' : v).trim(); });
  const values = hoja.getRange(2, 1, filas - 1, cols).getValues();

  return values.map(function(row) {
    const o = {};
    headers.forEach(function(h, i) {
      if (!h) return;
      o[h] = convertirValorMapaDV_(row[i]);
    });
    return o;
  }).filter(function(o) {
    return Object.keys(o).some(function(k) {
      return String(o[k] == null ? '' : o[k]).trim() !== '';
    });
  });
}

function limpiarPruebasCircuito() {
  const ss = obtenerSpreadsheet_();
  const hojas = [
    'USUARIOS',
    'CHOFERES',
    'VEHICULOS',
    'SERVICIOS',
    'ASIGNACIONES',
    'RUTAS',
    'PUNTOS_RUTA',
    'INSPECCIONES',
    'VIAJES',
    'EVENTOS',
    'GPS',
    'EVIDENCIAS',
    'REPORTES',
    'MANTENIMIENTO'
  ];

  const idsPorHoja = {
    USUARIOS: 'ID_USUARIO',
    CHOFERES: 'ID_CHOFER',
    VEHICULOS: 'ID_VEHICULO',
    SERVICIOS: 'ID_SERVICIO',
    ASIGNACIONES: 'ID_ASIGNACION',
    RUTAS: 'ID_RUTA',
    PUNTOS_RUTA: 'ID_PUNTO',
    INSPECCIONES: 'ID_INSPECCION',
    VIAJES: 'ID_VIAJE',
    EVENTOS: 'ID_EVENTO',
    GPS: 'ID_GPS',
    EVIDENCIAS: 'ID_EVIDENCIA',
    REPORTES: 'ID_REPORTE',
    MANTENIMIENTO: 'ID_MANTENIMIENTO'
  };

  const eliminados = {};

  hojas.forEach(nombre => {
    const hoja = ss.getSheetByName(nombre);
    if (!hoja || hoja.getLastRow() < 2) {
      eliminados[nombre] = 0;
      return;
    }

    const headers = obtenerEncabezados_(hoja);
    const idx = headers.indexOf(idsPorHoja[nombre]);

    if (idx < 0) {
      eliminados[nombre] = 0;
      return;
    }

    const valores = hoja
      .getRange(2, idx + 1, hoja.getLastRow() - 1, 1)
      .getValues();

    let cantidad = 0;

    for (let i = valores.length - 1; i >= 0; i--) {
      if (String(valores[i][0] || '').startsWith('TEST-')) {
        hoja.deleteRow(i + 2);
        cantidad++;
      }
    }

    eliminados[nombre] = cantidad;
  });

  const resultado = {
    ok: true,
    eliminados: eliminados
  };

  Logger.log(JSON.stringify(resultado, null, 2));
  return resultado;
}

    return {
      ok: true,
      api: 'API_V1',
      version: API_V1.VERSION,

      timestamp: isoHora_()
    };

    return obtenerResumenPanel_();

  // Para las lecturas generales, usamos el dispatcher existente.
  // No duplicamos lógica de acceso a las hojas.
  return ejecutarGet_(mapearAccionLecturaAPI_(accion), parametros);
}

  datos.origen = aplicacion;

  // REGLA CRITICA:
  // Solamente DIA_VERDE puede registrar GPS.
  if (accion === 'registrarGPS' && aplicacion !== API_V1.APLICACIONES.DIA_VERDE) {
    throw new Error('ACCESO DENEGADO: solamente DIA_VERDE puede registrar GPS.');
  }

  // CHOFERES nunca puede enviar GPS.
  if (aplicacion === API_V1.APLICACIONES.CHOFERES &&
      accion === 'registrarGPS') {
    throw new Error('ACCESO DENEGADO: CHOFERES no utiliza GPS.');
  }

  return ejecutarPost_(mapearAccionEscrituraAPI_(accion), datos);
}

function normalizarAplicacionAPI_(aplicacion) {

function validarAplicacionAPI_(aplicacion) {
  const permitidas = Object.values(API_V1.APLICACIONES);

  if (permitidas.indexOf(aplicacion) < 0) {
    throw new Error(
      'Aplicacion no permitida: ' + aplicacion
    );
  }
}

function validarPermisoAPI_(aplicacion, accion, tipo) {
  tipo = String(tipo || '').trim().toUpperCase();

  const clave = tipo === 'LECTURA'
    ? 'LECTURAS'
    : tipo === 'ESCRITURA'
      ? 'ESCRITURAS'
      : tipo;

  const permisos = API_V1[clave];

  if (!permisos) {
    throw new Error('Tipo de permiso no válido: ' + tipo);
  }

  const grupo = permisos[aplicacion] || [];

  if (grupo.indexOf(accion) < 0) {
    throw new Error(
      'Acción no permitida. Aplicación=' +
      aplicacion +
      ' Tipo=' +
      tipo +
      ' Acción=' +
      accion
    );
  }
}

// HASH DE CONTRASEÑA
// ============================================================

function hashPasswordV1_(password) {
  const texto = String(password == null ? '' : password);

  const bytes = Utilities.computeDigest(
    Utilities.DigestAlgorithm.SHA_256,
    texto,
    Utilities.Charset.UTF_8
  );

  return bytes.map(function(b) {
    const v = b < 0 ? b + 256 : b;
    return ('0' + v.toString(16)).slice(-2);
  }).join('');
}

function autenticarUsuarioV1_(usuario, password, aplicacion) {
  usuario = String(usuario || '').trim();
  aplicacion = String(aplicacion || '').trim().toUpperCase();

  if (!usuario) {
    throw new Error('USUARIO_REQUERIDO');
  }

  if (!password) {
    throw new Error('CONTRASENA_REQUERIDA');
  }

  validarAplicacionAuthV1_(aplicacion);

  const usuarios = leerTabla_(CONFIG.HOJAS.USUARIOS);

  const fila = usuarios.find(function(u) {
    return String(u.USUARIO || '').trim().toLowerCase() ===
      usuario.toLowerCase();
  });

  if (!fila) {
    throw new Error('CREDENCIALES_INVALIDAS');
  }

  const estado = String(fila.ESTADO || '').trim().toUpperCase();

  if (estado !== 'ACTIVO') {
    throw new Error('USUARIO_INACTIVO');
  }

  const hash = hashPasswordV1_(password);

  // En V1 se exige que la hoja ya tenga contraseña SHA-256.
  if (String(fila.CONTRASENA || '') !== hash) {
    throw new Error('CREDENCIALES_INVALIDAS');
  }

  const rol = String(fila.ROL || '').trim().toUpperCase();
  const aplicacionUsuario =
    String(fila.APLICACION || '').trim().toUpperCase();

  validarAccesoRolAplicacionV1_(rol, aplicacion, aplicacionUsuario);

  const token = crearTokenSesionV1_();
  const ahora = Date.now();
  const expira = ahora + AUTH_V1.DURACION_SESION_MS;

  const sesion = {
    token: token,
    idUsuario: String(fila.ID_USUARIO || ''),
    usuario: String(fila.USUARIO || ''),
    nombre: String(fila.NOMBRE || ''),
    rol: rol,
    aplicacion: aplicacion,
    aplicacionUsuario: aplicacionUsuario,
    idChofer: String(fila.ID_CHOFER || ''),
    creado: ahora,
    expira: expira
  };

  PropertiesService
    .getScriptProperties()
    .setProperty(
      'SESION_V1_' + token,
      JSON.stringify(sesion)
    );

  // Evento de auditoría.
  registrarEventoAuthV1_(
    'LOGIN',
    sesion,
    'Inicio de sesión correcto'
  );

  return {
    ok: true,
    token: token,
    usuario: {
      idUsuario: sesion.idUsuario,
      usuario: sesion.usuario,
      nombre: sesion.nombre,
      rol: sesion.rol,
      aplicacion: sesion.aplicacion,
      idChofer: sesion.idChofer
    },
    expira: new Date(expira).toISOString()
  };
}


// ============================================================
// VALIDAR SESION
// ============================================================

function validarSesionV1_(token, aplicacion) {
  token = String(token || '').trim();
  aplicacion = String(aplicacion || '').trim().toUpperCase();

  if (!token) {
    throw new Error('SESION_REQUERIDA');
  }

  validarAplicacionAuthV1_(aplicacion);

  const props = PropertiesService.getScriptProperties();
  const raw = props.getProperty('SESION_V1_' + token);

  if (!raw) {
    throw new Error('SESION_INVALIDA');
  }

  let sesion;

  try {
    sesion = JSON.parse(raw);
  } catch (err) {
    props.deleteProperty('SESION_V1_' + token);
    throw new Error('SESION_INVALIDA');
  }

  if (!sesion.expira || Date.now() >= Number(sesion.expira)) {
    props.deleteProperty('SESION_V1_' + token);

    try {
      registrarEventoAuthV1_(
        'LOGOUT',
        sesion,
        'Sesión expirada'
      );
    } catch (err) {}

    throw new Error('SESION_EXPIRADA');
  }

  if (String(sesion.aplicacion || '').toUpperCase() !== aplicacion) {
    throw new Error('APLICACION_NO_AUTORIZADA');
  }

  return sesion;
}


// ============================================================
// CERRAR SESION
// ============================================================

function cerrarSesionV1_(token) {
  token = String(token || '').trim();

  if (!token) {
    return {
      ok: true,
      cerrado: false
    };
  }

  const props = PropertiesService.getScriptProperties();
  const raw = props.getProperty('SESION_V1_' + token);

  if (raw) {
    try {
      const sesion = JSON.parse(raw);

      registrarEventoAuthV1_(
        'LOGOUT',
        sesion,
        'Cierre de sesión'
      );
    } catch (err) {}
  }

  props.deleteProperty('SESION_V1_' + token);

  return {
    ok: true,
    cerrado: true
  };
}


// ============================================================
// AUTORIZACION POR ROL
// ============================================================

function validarAccesoRolAplicacionV1_(
  rol,
  aplicacionSolicitada,
  aplicacionUsuario
) {
  rol = String(rol || '').trim().toUpperCase();
  aplicacionSolicitada =
    String(aplicacionSolicitada || '').trim().toUpperCase();
  aplicacionUsuario =
    String(aplicacionUsuario || '').trim().toUpperCase();

  if (!AUTH_V1.ROLES[rol]) {
    throw new Error('ROL_INVALIDO');
  }

  // ADMIN y SUPERVISOR trabajan desde PANEL.
  if (rol === AUTH_V1.ROLES.ADMIN ||
      rol === AUTH_V1.ROLES.SUPERVISOR) {

    if (aplicacionSolicitada !== AUTH_V1.APLICACIONES.PANEL) {
      throw new Error('ROL_NO_AUTORIZADO_PARA_APLICACION');
    }

    if (aplicacionUsuario &&
        aplicacionUsuario !== AUTH_V1.APLICACIONES.PANEL) {
      throw new Error('USUARIO_ASIGNADO_A_OTRA_APLICACION');
    }

    return true;
  }

  // CHOFER trabaja en CHOFERES o DIA_VERDE.
  if (rol === AUTH_V1.ROLES.CHOFER) {
    if (aplicacionSolicitada !== AUTH_V1.APLICACIONES.CHOFERES &&
        aplicacionSolicitada !== AUTH_V1.APLICACIONES.DIA_VERDE) {
      throw new Error('ROL_NO_AUTORIZADO_PARA_APLICACION');
    }

    if (aplicacionUsuario &&
        aplicacionUsuario !== aplicacionSolicitada) {
      throw new Error('USUARIO_ASIGNADO_A_OTRA_APLICACION');
    }

    return true;
  }

  throw new Error('ACCESO_DENEGADO');
}


function validarAplicacionAuthV1_(aplicacion) {
  const permitidas = Object.values(AUTH_V1.APLICACIONES);

  if (permitidas.indexOf(aplicacion) < 0) {
    throw new Error('APLICACION_INVALIDA');
  }
}

function obtenerUsuarioSesionV1_(token, aplicacion) {

      usuario: sesion.usuario,
      nombre: sesion.nombre,
      rol: sesion.rol,
      aplicacion: sesion.aplicacion,
      idChofer: sesion.idChofer
    },
    expira: new Date(Number(sesion.expira)).toISOString()
  };

// ============================================================
// EVENTO DE AUDITORIA
// ============================================================
// No depende del dispatcher de EVENTOS para evitar ciclos durante
// el login. Escribe directamente en EVENTOS.
//
// Si la hoja no está disponible, no debe romper el login.
// ============================================================

function registrarEventoAuthV1_(tipoEvento, sesion, descripcion) {
  try {
    const ss = obtenerSpreadsheet_();
    const hoja = ss.getSheetByName(CONFIG.HOJAS.EVENTOS);

    if (!hoja) return;

    const headers = obtenerEncabezados_(hoja);

    const fila = headers.map(function(h) {
      switch (h) {
        case 'ID_EVENTO':
          return Utilities.getUuid();

        case 'FECHA_HORA':
          return new Date();

        case 'TIPO_EVENTO':
          return tipoEvento;

        case 'ORIGEN':
          return sesion && sesion.aplicacion
            ? sesion.aplicacion
            : 'BACKEND';

        case 'ID_USUARIO':
          return sesion && sesion.idUsuario
            ? sesion.idUsuario
            : '';

        case 'ID_SERVICIO':
        case 'ID_ASIGNACION':
        case 'ID_VIAJE':
        case 'ID_VEHICULO':
        case 'ID_CHOFER':
          return '';

        case 'DESCRIPCION':
          return descripcion || '';

        case 'DATOS':
          return JSON.stringify({
            version: AUTH_V1.VERSION,
            rol: sesion && sesion.rol
              ? sesion.rol
              : ''
          });

        default:
          return '';
      }
    });

    hoja.appendRow(fila);
  } catch (err) {
    // La auditoría no debe impedir el funcionamiento de autenticación.
  }
}

  if (!passwordTest) throw new Error('Para crear o probar TEST_AUTH, configurar AUTH_TEST_PASSWORD en propiedades privadas del proyecto.');

  const existe = usuarios.some(function(u) {

      usuario: usuarioTest,
      password: passwordTest,
      mensaje: 'El usuario de prueba ya existía.'
    };
  }

  const valores = {

    USUARIO: usuarioTest,

    ESTADO: 'ACTIVO',

  };

  });

  hoja.appendRow(fila);

  return {

    creado: true,
    usuario: usuarioTest,
    password: passwordTest
  };

// ============================================================
// PRUEBA COMPLETA DE AUTENTICACION
// ============================================================
// 1) prepara usuario TEST
// 2) hace login
// 3) valida sesión
// 4) obtiene identidad
// 5) cierra sesión
// 6) confirma que la sesión quedó invalidada
//
// NO deja una sesión activa.
// ============================================================

function pruebaAutenticacionV1() {
  const resultado = {
    ok: false,
    pasos: []
  };

  try {
    const preparado = prepararUsuarioPruebaAuthV1();

    resultado.pasos.push({
      paso: 'USUARIO_TEST',
      ok: preparado.ok,
      creado: preparado.creado
    });

    const login = autenticarUsuarioV1_(
      preparado.usuario,
      preparado.password,
      'PANEL'
    );

    resultado.pasos.push({
      paso: 'LOGIN',
      ok: login.ok,
      rol: login.usuario.rol,
      aplicacion: login.usuario.aplicacion
    });

    const sesion = validarSesionV1_(
      login.token,
      'PANEL'
    );

    resultado.pasos.push({
      paso: 'VALIDAR_SESION',
      ok: true,
      idUsuario: sesion.idUsuario
    });

    const identidad = obtenerUsuarioSesionV1_(
      login.token,
      'PANEL'
    );

    resultado.pasos.push({
      paso: 'IDENTIDAD',
      ok: identidad.ok,
      usuario: identidad.usuario.usuario,
      rol: identidad.usuario.rol
    });

    const cierre = cerrarSesionV1_(login.token);

    resultado.pasos.push({
      paso: 'LOGOUT',
      ok: cierre.ok && cierre.cerrado
    });

    let invalidada = false;

    try {
      validarSesionV1_(
        login.token,
        'PANEL'
      );
    } catch (err) {
      invalidada = true;
    }

    resultado.pasos.push({
      paso: 'SESION_INVALIDADA',
      ok: invalidada
    });

    resultado.ok = resultado.pasos.every(function(p) {
      return p.ok === true;
    });

    Logger.log(
      JSON.stringify(resultado, null, 2)
    );

    return resultado;

  } catch (err) {
    resultado.error =
      err && err.message
        ? err.message
        : String(err);

    Logger.log(
      JSON.stringify(resultado, null, 2)
    );

    return resultado;
  }
}

  };

  function probarCaso(nombre, rol, appUsuario, appSolicitada, esperado) {

        appSolicitada,
        appUsuario
      );
      permitido = true;
    } catch (err) {
      permitido = false;
    }

    const correcto = permitido === esperado;

    resultado.pruebas.push({
      caso: nombre,
      permitido: permitido,
      esperado: esperado,
      ok: correcto
    });

    if (!correcto) {
      resultado.ok = false;
    }
  }

  probarCaso(
    'ADMIN_PANEL_EN_PANEL',
    'ADMIN',
    'PANEL',
    'PANEL',
    true
  );

  probarCaso(
    'ADMIN_PANEL_EN_DIA_VERDE',
    'ADMIN',
    'PANEL',
    'DIA_VERDE',
    false

function sv_getSpreadsheet_() {
  return SpreadsheetApp.getActiveSpreadsheet();
}

function sv_getSheet_(nombre) {
  const ss = sv_getSpreadsheet_();
  const sh = ss.getSheetByName(nombre);

  if (!sh) {
    throw new Error('No existe la hoja requerida: ' + nombre);
  }

  return sh;
}

function sv_getData_(nombreHoja) {
  const sh = sv_getSheet_(nombreHoja);
  const lastRow = sh.getLastRow();
  const lastCol = sh.getLastColumn();

  if (lastRow < 1 || lastCol < 1) {
    return { headers: [], rows: [] };
  }

  const values = sh.getRange(1, 1, lastRow, lastCol).getValues();

  return {
    headers: values[0].map(String),
    rows: values.slice(1)
  };
}

function sv_assertHeaders_(nombreHoja, esperadas) {
  const data = sv_getData_(nombreHoja);

  if (data.headers.length !== esperadas.length) {
    throw new Error(
      'La hoja ' + nombreHoja +
      ' no coincide con la estructura esperada. ' +
      'Esperadas: ' + esperadas.join('|') +
      ' / Encontradas: ' + data.headers.join('|')
    );
  }

  for (let i = 0; i < esperadas.length; i++) {
    if (String(data.headers[i]).trim() !== esperadas[i]) {
      throw new Error(
        'La hoja ' + nombreHoja +
        ' tiene una columna incorrecta en posición ' + (i + 1) +
        '. Esperada: ' + esperadas[i] +
        ' / Encontrada: ' + data.headers[i]
      );
    }
  }

  return true;
}

function sv_rowToObject_(headers, row) {
  const obj = {};

  headers.forEach(function(h, i) {
    obj[h] = row[i];
  });

  return obj;
}

function sv_normalizarTexto_(valor) {
  return valor === null || valor === undefined
    ? ''
    : String(valor).trim();
}

function sv_generarId_(prefijo) {
  const ahora = new Date();

  return prefijo + '-' +
    Utilities.formatDate(
      ahora,
      Session.getScriptTimeZone() || 'America/Argentina/Buenos_Aires',
      'yyyyMMdd-HHmmss'
    ) +
    '-' +
    Utilities.getUuid().substring(0, 8).toUpperCase();
}

function sv_fechaClave_(valor) {
  if (valor instanceof Date && !isNaN(valor.getTime())) {
    return Utilities.formatDate(
      valor,
      Session.getScriptTimeZone() || 'America/Argentina/Buenos_Aires',
      'yyyy-MM-dd'
    );
  }

  const s = sv_normalizarTexto_(valor);

  if (!s) return '';

  return s.substring(0, 10);
}

function sv_ahora_() {
  return new Date();
}

function sv_registrarEvento_(tipoEvento, origen, datos) {
  try {
    const sh = sv_getSheet_(SV_CONFIG.HOJA_EVENTOS);
    const headers = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0];

    const requerido = [
      'ID_EVENTO',
      'FECHA_HORA',
      'TIPO_EVENTO',
      'ORIGEN',
      'ID_USUARIO',
      'ID_SERVICIO',
      'ID_ASIGNACION',
      'ID_VIAJE',
      'ID_VEHICULO',
      'ID_CHOFER',
      'DESCRIPCION',
      'DATOS'
    ];

    for (let i = 0; i < requerido.length; i++) {
      if (headers[i] !== requerido[i]) {
        return false;
      }
    }

    const fila = headers.map(function(h) {
      switch (h) {
        case 'ID_EVENTO':
          return sv_generarId_('EVT');

        case 'FECHA_HORA':
          return sv_ahora_();

        case 'TIPO_EVENTO':
          return tipoEvento;

        case 'ORIGEN':
          return origen;

        case 'ID_USUARIO':
          return datos.ID_USUARIO || '';

        case 'ID_SERVICIO':
          return datos.ID_SERVICIO || '';

        case 'ID_ASIGNACION':
          return datos.ID_ASIGNACION || '';

        case 'ID_VIAJE':
          return datos.ID_VIAJE || '';

        case 'ID_VEHICULO':
          return datos.ID_VEHICULO || '';

        case 'ID_CHOFER':
          return datos.ID_CHOFER || '';

        case 'DESCRIPCION':
          return datos.DESCRIPCION || '';

        case 'DATOS':
          return JSON.stringify(datos.DATOS || {});

        default:
          return '';
      }
    });

    sh.appendRow(fila);
    return true;

  } catch (err) {
    console.warn('No se pudo registrar EVENTO: ' + err.message);
    return false;
  }
}

function sv_buscarPorId_(nombreHoja, nombreId, id) {
  const data = sv_getData_(nombreHoja);

  for (let i = 0; i < data.rows.length; i++) {
    const obj = sv_rowToObject_(data.headers, data.rows[i]);

    if (sv_normalizarTexto_(obj[nombreId]) === sv_normalizarTexto_(id)) {
      return {
        index: i + 2,
        objeto: obj
      };
    }
  }

  return null;
}

function sv_validarChofer_(idChofer) {
  const encontrado = sv_buscarPorId_(
    SV_CONFIG.HOJA_CHOFERES,
    'ID_CHOFER',
    idChofer
  );

  if (!encontrado) {
    throw new Error('El chofer no existe: ' + idChofer);
  }

  const estado = sv_normalizarTexto_(encontrado.objeto.ESTADO).toUpperCase();

  if (estado && estado !== 'ACTIVO') {
    throw new Error(
      'El chofer no está ACTIVO. Estado actual: ' + estado
    );
  }

  return encontrado.objeto;
}

function sv_validarVehiculo_(idVehiculo) {
  const encontrado = sv_buscarPorId_(
    SV_CONFIG.HOJA_VEHICULOS,
    'ID_VEHICULO',
    idVehiculo
  );

  if (!encontrado) {
    throw new Error('El vehículo no existe: ' + idVehiculo);
  }

  const activo = sv_normalizarTexto_(
    encontrado.objeto.ACTIVO
  ).toUpperCase();

  const estado = sv_normalizarTexto_(
    encontrado.objeto.ESTADO_OPERATIVO
  ).toUpperCase();

  if (activo && !['SI', 'TRUE', '1', 'ACTIVO'].includes(activo)) {
    throw new Error(
      'El vehículo no está ACTIVO. Valor ACTIVO: ' + activo
    );
  }

  const estadosBloqueados = [
    'BAJA',
    'INACTIVO',
    'FUERA DE SERVICIO',
    'RADIADO'
  ];

  if (estadosBloqueados.includes(estado)) {
    throw new Error(
      'El vehículo no está disponible. Estado operativo: ' + estado
    );
  }

  return encontrado.objeto;
}

function sv_crearServicio(datos) {
  sv_assertHeaders_(
    SV_CONFIG.HOJA_SERVICIOS,
    SV_HEADERS_SERVICIOS
  );

  datos = datos || {};

  const fecha = sv_normalizarTexto_(datos.FECHA);
  const hora = sv_normalizarTexto_(datos.HORA_PROGRAMADA);
  const tipo = sv_normalizarTexto_(datos.TIPO_SERVICIO);
  const nombre = sv_normalizarTexto_(datos.NOMBRE_SERVICIO);
  const idRuta = sv_normalizarTexto_(datos.ID_RUTA);
  const observaciones = sv_normalizarTexto_(datos.OBSERVACIONES);

  if (!fecha) throw new Error('FECHA es obligatoria.');
  if (!tipo) throw new Error('TIPO_SERVICIO es obligatorio.');
  if (!nombre) throw new Error('NOMBRE_SERVICIO es obligatorio.');

  const idServicio = datos.ID_SERVICIO
    ? sv_normalizarTexto_(datos.ID_SERVICIO)
    : sv_generarId_('SER');

  if (sv_obtenerServicioInterno_(idServicio)) {
    throw new Error('Ya existe el ID_SERVICIO: ' + idServicio);
  }

  const sh = sv_getSheet_(SV_CONFIG.HOJA_SERVICIOS);

  sh.appendRow([
    idServicio,
    fecha,
    hora,
    tipo,
    nombre,
    idRuta,
    '',
    'PROGRAMADO',
    '',
    '',
    observaciones
  ]);

  sv_registrarEvento_(
    'SERVICIO_CREADO',
    'PANEL',
    {
      ID_SERVICIO: idServicio,
      DESCRIPCION: 'Servicio creado desde PANEL',
      DATOS: {
        FECHA: fecha,
        HORA_PROGRAMADA: hora,
        TIPO_SERVICIO: tipo,
        NOMBRE_SERVICIO: nombre,
        ID_RUTA: idRuta
      }
    }
  );

  return {
    ok: true,
    idServicio: idServicio,
    estado: 'PROGRAMADO',
    mensaje: 'Servicio creado correctamente.'
  };
}

function sv_crearAsignacion(datos) {
  sv_assertHeaders_(
    SV_CONFIG.HOJA_ASIGNACIONES,
    SV_HEADERS_ASIGNACIONES
  );

  datos = datos || {};

  const fecha = sv_normalizarTexto_(datos.FECHA);
  const idServicio = sv_normalizarTexto_(datos.ID_SERVICIO);
  const idChofer = sv_normalizarTexto_(datos.ID_CHOFER);
  const idVehiculo = sv_normalizarTexto_(datos.ID_VEHICULO);
  const turno = sv_normalizarTexto_(datos.TURNO);
  const observaciones = sv_normalizarTexto_(datos.OBSERVACIONES);

  if (!fecha) throw new Error('FECHA es obligatoria.');
  if (!idServicio) throw new Error('ID_SERVICIO es obligatorio.');
  if (!idChofer) throw new Error('ID_CHOFER es obligatorio.');
  if (!idVehiculo) throw new Error('ID_VEHICULO es obligatorio.');

  const servicio = sv_obtenerServicioInterno_(idServicio);

  if (!servicio) {
    throw new Error('No existe el servicio: ' + idServicio);
  }

  const estadoServicio =
    sv_normalizarTexto_(servicio.objeto.ESTADO).toUpperCase();

  if (['FINALIZADO', 'CANCELADO'].includes(estadoServicio)) {
    throw new Error(
      'No se puede asignar un servicio con estado ' +
      estadoServicio
    );
  }

  const fechaServicio = sv_fechaClave_(servicio.objeto.FECHA);
  const fechaAsignacion = sv_fechaClave_(fecha);

  if (
    fechaServicio &&
    fechaAsignacion &&
    fechaServicio !== fechaAsignacion
  ) {
    throw new Error(
      'La FECHA de la asignación no coincide con la FECHA del servicio.'
    );
  }

  const chofer = sv_validarChofer_(idChofer);
  const vehiculo = sv_validarVehiculo_(idVehiculo);

  const data = sv_getData_(SV_CONFIG.HOJA_ASIGNACIONES);

  for (let i = 0; i < data.rows.length; i++) {
    const r = sv_rowToObject_(data.headers, data.rows[i]);

    const estado = sv_normalizarTexto_(r.ESTADO).toUpperCase();

    if (
      sv_fechaClave_(r.FECHA) === fechaAsignacion &&
      (
        sv_normalizarTexto_(r.ID_CHOFER) === idChofer ||
        sv_normalizarTexto_(r.ID_VEHICULO) === idVehiculo
      ) &&
      !['CANCELADA', 'FINALIZADA'].includes(estado)
    ) {
      throw new Error(
        'El chofer o el vehículo ya tiene una asignación activa para esa fecha.'
      );
    }
  }

  const idAsignacion = datos.ID_ASIGNACION
    ? sv_normalizarTexto_(datos.ID_ASIGNACION)
    : sv_generarId_('ASG');

  if (sv_obtenerAsignacionInterno_(idAsignacion)) {
    throw new Error('Ya existe el ID_ASIGNACION: ' + idAsignacion);
  }

  const ahora = sv_ahora_();

  const sh = sv_getSheet_(SV_CONFIG.HOJA_ASIGNACIONES);

  sh.appendRow([
    idAsignacion,
    fecha,
    idServicio,
    idChofer,
    idVehiculo,
    turno,
    'CONFIRMADA',
    ahora,
    observaciones
  ]);

  // Actualizamos el vínculo en SERVICIOS.
  const colIdAsignacion = SV_HEADERS_SERVICIOS.indexOf('ID_ASIGNACION') + 1;
  const filaServicio = servicio.index;

  sv_getSheet_(SV_CONFIG.HOJA_SERVICIOS)
    .getRange(filaServicio, colIdAsignacion)
    .setValue(idAsignacion);

  sv_registrarEvento_(
    'ASIGNACION_CREADA',
    'PANEL',
    {
      ID_SERVICIO: idServicio,
      ID_ASIGNACION: idAsignacion,
      ID_CHOFER: idChofer,
      ID_VEHICULO: idVehiculo,
      DESCRIPCION: 'Asignación creada desde PANEL',
      DATOS: {
        FECHA: fecha,
        TURNO: turno,
        CHOFER: chofer.NOMBRE || '',
        VEHICULO: vehiculo.DOMINIO || ''
      }
    }
  );

  return {
    ok: true,
    idAsignacion: idAsignacion,
    idServicio: idServicio,
    idChofer: idChofer,
    idVehiculo: idVehiculo,
    estado: 'CONFIRMADA',
    mensaje: 'Asignación creada correctamente.'
  };
}

function sv_actualizarEstadoAsignacion(
  idAsignacion,
  estado,
  observaciones
) {
  const estadosPermitidos = [
    'PENDIENTE',
    'CONFIRMADA',
    'EN SERVICIO',
    'FINALIZADA',
    'CANCELADA'
  ];

  estado = sv_normalizarTexto_(estado).toUpperCase();

  if (!estadosPermitidos.includes(estado)) {
    throw new Error(
      'Estado de asignación no permitido: ' + estado
    );
  }

  const encontrado =
    sv_obtenerAsignacionInterno_(idAsignacion);

  if (!encontrado) {
    throw new Error(
      'Asignación no encontrada: ' + idAsignacion
    );
  }

  const sh = sv_getSheet_(SV_CONFIG.HOJA_ASIGNACIONES);

  const colEstado =
    SV_HEADERS_ASIGNACIONES.indexOf('ESTADO') + 1;

  const colObs =
    SV_HEADERS_ASIGNACIONES.indexOf('OBSERVACIONES') + 1;

  sh.getRange(encontrado.index, colEstado)
    .setValue(estado);

  if (observaciones !== undefined) {
    sh.getRange(encontrado.index, colObs)
      .setValue(sv_normalizarTexto_(observaciones));
  }

  sv_registrarEvento_(
    'ASIGNACION_ESTADO_CAMBIADO',
    'PANEL',
    {
      ID_SERVICIO: encontrado.objeto.ID_SERVICIO || '',
      ID_ASIGNACION: idAsignacion,
      ID_CHOFER: encontrado.objeto.ID_CHOFER || '',
      ID_VEHICULO: encontrado.objeto.ID_VEHICULO || '',
      DESCRIPCION: 'Cambio de estado de asignación',
      DATOS: {
        NUEVO_ESTADO: estado,
        OBSERVACIONES: observaciones || ''
      }
    }
  );

  return {
    ok: true,
    idAsignacion: idAsignacion,
    estado: estado
  };
}

  const sh = sv_getSheet_(nombreHoja);

  for (let i = data.rows.length - 1; i >= 0; i--) {
    const obj =
      sv_rowToObject_(data.headers, data.rows[i]);

    if (
      sv_normalizarTexto_(obj[nombreId]) ===
      sv_normalizarTexto_(id)
    ) {
      sh.deleteRow(i + 2);
      return true;
    }
  }

  return false;
}

    .getScriptProperties()

function IV1_obtenerHoja_(nombre) {
  const ss = IV1_obtenerSpreadsheet_();
  const hoja = ss.getSheetByName(nombre);
  if (!hoja) throw new Error('No existe la hoja: ' + nombre);
  return hoja;
}

function IV1_encabezados_(nombre) {
  const hoja = IV1_obtenerHoja_(nombre);
  const ultimaCol = hoja.getLastColumn();
  if (ultimaCol < 1) return [];
  return hoja.getRange(1, 1, 1, ultimaCol).getValues()[0].map(String);
}

function IV1_indice_(nombre, campo) {
  const headers = IV1_encabezados_(nombre);
  const i = headers.indexOf(campo);
  if (i === -1) throw new Error(
    'La hoja ' + nombre + ' no contiene la columna ' + campo
  );
  return i;
}

function IV1_registros_(nombre) {
  const hoja = IV1_obtenerHoja_(nombre);
  const ultimaFila = hoja.getLastRow();
  const ultimaCol = hoja.getLastColumn();
  if (ultimaFila < 2 || ultimaCol < 1) return [];
  return hoja.getRange(2, 1, ultimaFila - 1, ultimaCol).getValues();
}

function IV1_buscarPorId_(nombre, campoId, id) {
  const headers = IV1_encabezados_(nombre);
  const idx = headers.indexOf(campoId);
  if (idx === -1) throw new Error(
    'La hoja ' + nombre + ' no contiene ' + campoId
  );

  const filas = IV1_registros_(nombre);
  for (let i = 0; i < filas.length; i++) {
    if (String(filas[i][idx]) === String(id)) {
      return {
        hoja: IV1_obtenerHoja_(nombre),
        fila: i + 2,
        headers: headers,
        valores: filas[i]
      };
    }
  }
  return null;
}

function IV1_id_(prefijo) {
  return 'TEST-' + prefijo + '-' +
    Utilities.formatDate(new Date(), Session.getScriptTimeZone() || 'America/Argentina/Buenos_Aires', 'yyyyMMdd-HHmmss') +
    '-' + Utilities.getUuid().substring(0, 8).toUpperCase();
}

function IV1_agregar_(nombre, objeto) {
  const hoja = IV1_obtenerHoja_(nombre);
  const headers = IV1_encabezados_(nombre);
  const fila = headers.map(function(h) {
    return Object.prototype.hasOwnProperty.call(objeto, h) ? objeto[h] : '';
  });
  hoja.appendRow(fila);
  return objeto;
}

function IV1_actualizarCampos_(nombre, campoId, id, cambios) {
  const encontrado = IV1_buscarPorId_(nombre, campoId, id);
  if (!encontrado) throw new Error(
    'No se encontró ' + campoId + '=' + id + ' en ' + nombre
  );

  const headers = encontrado.headers;
  const valores = encontrado.valores.slice();

  Object.keys(cambios || {}).forEach(function(campo) {
    const idx = headers.indexOf(campo);
    if (idx === -1) throw new Error(
      'La hoja ' + nombre + ' no contiene la columna ' + campo
    );
    valores[idx] = cambios[campo];
  });

  encontrado.hoja.getRange(encontrado.fila, 1, 1, headers.length).setValues([valores]);
  return true;
}

function IV1_crearInspeccion_(datos) {
  if (!datos || !datos.idAsignacion) {
    throw new Error('Falta ID_ASIGNACION.');
  }
  if (!datos.idVehiculo) throw new Error('Falta ID_VEHICULO.');
  if (!datos.idChofer) throw new Error('Falta ID_CHOFER.');

  const asignacion = IV1_buscarPorId_(
    'ASIGNACIONES',
    'ID_ASIGNACION',
    datos.idAsignacion
  );
  if (!asignacion) throw new Error('La asignación no existe.');

  const vehiculo = IV1_buscarPorId_(
    'VEHICULOS',
    'ID_VEHICULO',
    datos.idVehiculo
  );
  if (!vehiculo) throw new Error('El vehículo no existe.');

  const chofer = IV1_buscarPorId_(
    'CHOFERES',
    'ID_CHOFER',
    datos.idChofer
  );
  if (!chofer) throw new Error('El chofer no existe.');

  const estadoAsignacion = String(
    asignacion.valores[IV1_indice_('ASIGNACIONES', 'ESTADO')]
  ).toUpperCase();

  if (estadoAsignacion === 'FINALIZADA' || estadoAsignacion === 'CANCELADA') {
    throw new Error('La asignación no permite iniciar una inspección.');
  }

  const ahora = new Date();
  const id = datos.idInspeccion || IV1_id_('INS');

  const objeto = {
    ID_INSPECCION: id,
    FECHA: Utilities.formatDate(
      ahora,
      Session.getScriptTimeZone() || 'America/Argentina/Buenos_Aires',
      'yyyy-MM-dd'
    ),
    HORA: Utilities.formatDate(
      ahora,
      Session.getScriptTimeZone() || 'America/Argentina/Buenos_Aires',
      'HH:mm:ss'
    ),
    ID_ASIGNACION: datos.idAsignacion,
    ID_VEHICULO: datos.idVehiculo,
    ID_CHOFER: datos.idChofer,
    KM_INICIAL: datos.kmInicial !== undefined ? datos.kmInicial : '',
    RESULTADO: datos.resultado || 'PENDIENTE',
    OBSERVACIONES: datos.observaciones || '',
    EVIDENCIA_ID: datos.evidenciaId || '',
    ESTADO: 'EN CURSO',
    FECHA_FINALIZACION: ''
  };

  IV1_agregar_('INSPECCIONES', objeto);
  return objeto;
}

// ------------------------------------------------------------
// FINALIZAR INSPECCION
// ------------------------------------------------------------
function IV1_finalizarInspeccion_(idInspeccion, resultado, observaciones) {
  if (!idInspeccion) throw new Error('Falta ID_INSPECCION.');

  const permitido = ['APROBADA', 'OBSERVADA', 'RECHAZADA'];
  const r = String(resultado || '').toUpperCase();

  if (permitido.indexOf(r) === -1) {
    throw new Error(
      'RESULTADO inválido. Use APROBADA, OBSERVADA o RECHAZADA.'
    );
  }

  const encontrada = IV1_buscarPorId_(
    'INSPECCIONES',
    'ID_INSPECCION',
    idInspeccion
  );
  if (!encontrada) throw new Error('La inspección no existe.');

  const estadoActual = String(
    encontrada.valores[IV1_indice_('INSPECCIONES', 'ESTADO')]
  ).toUpperCase();

  if (estadoActual === 'FINALIZADA') {
    throw new Error('La inspección ya está finalizada.');
  }

  const ahora = new Date();

  IV1_actualizarCampos_(
    'INSPECCIONES',
    'ID_INSPECCION',
    idInspeccion,
    {
      RESULTADO: r,
      OBSERVACIONES: observaciones || '',
      ESTADO: 'FINALIZADA',
      FECHA_FINALIZACION: ahora
    }
  );

  return {
    ok: true,
    idInspeccion: idInspeccion,
    resultado: r,
    estado: 'FINALIZADA'
  };
}

// CONSULTAR INSPECCION
// ------------------------------------------------------------
function IV1_obtenerInspeccion_(idInspeccion) {
  const encontrada = IV1_buscarPorId_(
    'INSPECCIONES',
    'ID_INSPECCION',
    idInspeccion
  );

  if (!encontrada) {
    return { ok: false, error: 'INSPECCION_NO_ENCONTRADA' };
  }

  const objeto = {};
  encontrada.headers.forEach(function(h, i) {
    objeto[h] = encontrada.valores[i];
  });

  return { ok: true, inspeccion: objeto };
}

// LIMPIEZA DE PRUEBAS
// SOLO elimina registros cuyo ID comience con TEST-.

function limpiarPruebasInspeccionesV1() {
  const hojas = [
    ['INSPECCIONES', 'ID_INSPECCION'],
    ['ASIGNACIONES', 'ID_ASIGNACION'],
    ['SERVICIOS', 'ID_SERVICIO'],
    ['CHOFERES', 'ID_CHOFER'],
    ['VEHICULOS', 'ID_VEHICULO']
  ];

  const eliminados = {};

  hojas.forEach(function(par) {
    const nombreHoja = par[0];
    const campoId = par[1];
    const hoja = IV1_obtenerHoja_(nombreHoja);
    const headers = IV1_encabezados_(nombreHoja);
    const idx = headers.indexOf(campoId);

    if (idx === -1) {
      eliminados[nombreHoja] = 0;
      return;
    }

    const ultimaFila = hoja.getLastRow();
    if (ultimaFila < 2) {
      eliminados[nombreHoja] = 0;
      return;
    }

    const valores = hoja.getRange(2, 1, ultimaFila - 1, hoja.getLastColumn()).getValues();
    const filasAEliminar = [];

    valores.forEach(function(fila, i) {
      if (String(fila[idx]).indexOf('TEST-') === 0) {
        filasAEliminar.push(i + 2);
      }
    });

    filasAEliminar.reverse().forEach(function(fila) {
      hoja.deleteRow(fila);
    });

    eliminados[nombreHoja] = filasAEliminar.length;
  });

  const resultado = {
    ok: true,
    eliminados: eliminados
  };

  Logger.log(JSON.stringify(resultado, null, 2));
  return resultado;
}

const FCM_CONFIG = {
  PROJECT_ID: 'panel-de-control-ac0d8',
  ENDPOINT: 'https://fcm.googleapis.com/v1/projects/panel-de-control-ac0d8/messages:send',
  TOKEN_ENDPOINT: 'https://oauth2.googleapis.com/token',
  SERVICE_ACCOUNT_PROPERTY: 'FCM_SERVICE_ACCOUNT_JSON'
};

const PUSH_HEADERS = [
  'ID_PUSH','ID_CHOFER','TOKEN_FCM','PLATAFORMA','ACTIVO',
  'FECHA_ALTA','FECHA_ULTIMA_ACTUALIZACION','ULTIMO_ENVIO',
  'ULTIMO_ERROR','APP_URL','USER_AGENT'
];

function asegurarHojaNotificacionesPush_(ss) {
  ss = ss || obtenerSpreadsheet_();
  let hoja = ss.getSheetByName(CONFIG.HOJAS.NOTIFICACIONES_PUSH);
  if (!hoja) {
    hoja = ss.insertSheet(CONFIG.HOJAS.NOTIFICACIONES_PUSH);
    hoja.getRange(1,1,1,PUSH_HEADERS.length).setValues([PUSH_HEADERS]);
    hoja.setFrozenRows(1);
  } else {
    const actuales = obtenerEncabezados_(hoja);
    const faltantes = PUSH_HEADERS.filter(h => actuales.indexOf(h) < 0);
    if (faltantes.length) {
      throw new Error('La hoja NOTIFICACIONES_PUSH existe pero no tiene las columnas requeridas: '+faltantes.join(', '));
    }
  }
  return hoja;
}

function registrarTokenPush_(d) {
  validarCampos_(d,['idChofer','token']);
  const token = String(d.token || '').trim();
  if (!token) throw new Error('TOKEN_FCM vacío.');

  const hoja = asegurarHojaNotificacionesPush_();
  const filas = leerTabla_(CONFIG.HOJAS.NOTIFICACIONES_PUSH);
  const existente = filas.find(r => String(r.TOKEN_FCM || '').trim() === token);
  const ahora = ahora_();
  const datos = {
    ID_CHOFER: String(d.idChofer || '').trim(),
    TOKEN_FCM: token,
    PLATAFORMA: String(d.plataforma || 'WEB').trim(),
    ACTIVO: 'SI',
    FECHA_ULTIMA_ACTUALIZACION: ahora,
    APP_URL: String(d.appUrl || '').trim(),
    USER_AGENT: String(d.userAgent || '').slice(0,1000)
  };

  if (existente) {
    // Un token de navegador debe pertenecer a un solo chofer.
    // Esto evita notificar al chofer anterior si el teléfono cambia de sesión.
    const resultado = actualizarPorID_(CONFIG.HOJAS.NOTIFICACIONES_PUSH,'ID_PUSH',existente.ID_PUSH,datos);
    registrarEvento_({
      tipoEvento:'PUSH_TOKEN_ACTUALIZADO',
      origen:'CHOFERES',
      idChofer:datos.ID_CHOFER,
      descripcion:'Token FCM actualizado',
      datos:{idPush:existente.ID_PUSH,plataforma:datos.PLATAFORMA}
    });
    return {ok:true,accion:'actualizar',registro:resultado};
  }

  const registro = insertarObjeto_(CONFIG.HOJAS.NOTIFICACIONES_PUSH,{
    ID_PUSH: generarID_('PUSH'),
    ID_CHOFER: datos.ID_CHOFER,
    TOKEN_FCM: datos.TOKEN_FCM,
    PLATAFORMA: datos.PLATAFORMA,
    ACTIVO: 'SI',
    FECHA_ALTA: ahora,
    FECHA_ULTIMA_ACTUALIZACION: ahora,
    ULTIMO_ENVIO: '',
    ULTIMO_ERROR: '',
    APP_URL: datos.APP_URL,
    USER_AGENT: datos.USER_AGENT
  });

  registrarEvento_({
    tipoEvento:'PUSH_TOKEN_REGISTRADO',
    origen:'CHOFERES',
    idChofer:datos.ID_CHOFER,
    descripcion:'Dispositivo preparado para notificaciones push',
    datos:{idPush:registro.ID_PUSH,plataforma:datos.PLATAFORMA}
  });

  return {ok:true,accion:'registrar',registro:registro};
}

function desregistrarTokenPush_(d) {
  validarCampos_(d,['token']);
  const token = String(d.token || '').trim();

function enviarNotificacionAsignacion_(asignacion, tipoEvento) {
  try {
    const datos = obtenerDatosAsignacionPush_(asignacion);
    const tokens = obtenerTokensPushActivos_(datos.idChofer);
    if (!tokens.length) {
      registrarEvento_({
        tipoEvento:'PUSH_SIN_DISPOSITIVO',
        origen:'PANEL',
        idServicio:datos.idServicio,
        idAsignacion:datos.idAsignacion,
        idVehiculo:datos.idVehiculo,
        idChofer:datos.idChofer,
        descripcion:'No hay dispositivos push activos para el chofer',
        datos:{tipoPush:tipoEvento}
      });
      return {ok:true,enviados:0,motivo:'SIN_DISPOSITIVO'};
    }

    const titulo = tipoEvento === 'ASIGNACION_REASIGNADA'
      ? '🔄 Servicio reasignado'
      : '🔔 Nuevo servicio asignado';

    const partes = [datos.nombreServicio];
    if (datos.fecha) partes.push('Fecha: '+datos.fecha);
    if (datos.hora) partes.push('Hora: '+datos.hora);
    if (datos.dominio) partes.push('Unidad: '+datos.dominio);
    else if (datos.interno) partes.push('Interno: '+datos.interno);
    const cuerpo = partes.join(' · ');

    let enviados = 0;
    const errores = [];
    tokens.forEach(function(registro){
      try {
        const resultado = enviarPushFCM_(registro.TOKEN_FCM, titulo, cuerpo, {
          tipo: tipoEvento,
          idAsignacion: datos.idAsignacion,
          idServicio: datos.idServicio,
          idChofer: datos.idChofer,
          idVehiculo: datos.idVehiculo,
          fecha: datos.fecha,
          hora: datos.hora,
          dominio: datos.dominio,
          turno: datos.turno
        }, registro.APP_URL);

        actualizarPorID_(CONFIG.HOJAS.NOTIFICACIONES_PUSH,'ID_PUSH',registro.ID_PUSH,{
          ULTIMO_ENVIO:ahora_(),
          ULTIMO_ERROR:''
        });
        enviados++;
      } catch (err) {
        const mensaje = String(err && err.message ? err.message : err);
        errores.push({idPush:registro.ID_PUSH,error:mensaje});
        const cambios = {ULTIMO_ERROR:mensaje.slice(0,2000),FECHA_ULTIMA_ACTUALIZACION:ahora_()};
        if (/UNREGISTERED|NOT_FOUND|registration-token-not-registered/i.test(mensaje)) cambios.ACTIVO='NO';
        try { actualizarPorID_(CONFIG.HOJAS.NOTIFICACIONES_PUSH,'ID_PUSH',registro.ID_PUSH,cambios); } catch(e) {}
      }
    });

    registrarEvento_({
      tipoEvento:'PUSH_ASIGNACION_ENVIADA',
      origen:'PANEL',
      idServicio:datos.idServicio,
      idAsignacion:datos.idAsignacion,
      idVehiculo:datos.idVehiculo,
      idChofer:datos.idChofer,
      descripcion:'Notificación push procesada',
      datos:{tipoPush:tipoEvento,dispositivos:tokens.length,enviados:enviados,errores:errores.length}
    });

    return {ok:true,enviados:enviados,errores:errores};
  } catch (err) {
    registrarEvento_({
      tipoEvento:'PUSH_ERROR',
      origen:'PANEL',
      idServicio:asignacion?.ID_SERVICIO,
      idAsignacion:asignacion?.ID_ASIGNACION,
      idVehiculo:asignacion?.ID_VEHICULO,
      idChofer:asignacion?.ID_CHOFER,
      descripcion:'Error procesando notificación push',
      datos:{error:String(err && err.message ? err.message : err)}
    });
    return {ok:false,error:String(err && err.message ? err.message : err)};
  }
}

function actualizarAsignacionConNotificacion_(d) {
  validarCampos_(d,['idAsignacion']);
  const actual = obtenerPorID_(CONFIG.HOJAS.ASIGNACIONES,'ID_ASIGNACION',d.idAsignacion);
  if (!actual) throw new Error('Registro no encontrado: '+d.idAsignacion);
  const cambios = d.cambios || {};
  const nuevoChofer = String(cambios.ID_CHOFER || actual.ID_CHOFER || '').trim();
  const nuevoVehiculo = String(cambios.ID_VEHICULO || actual.ID_VEHICULO || '').trim();
  const cambioChofer = nuevoChofer !== String(actual.ID_CHOFER || '').trim();
  const cambioVehiculo = nuevoVehiculo !== String(actual.ID_VEHICULO || '').trim();

  const resultado = actualizarPorID_(CONFIG.HOJAS.ASIGNACIONES,'ID_ASIGNACION',d.idAsignacion,cambios);

  if (cambioChofer || cambioVehiculo) {
    if (cambioChofer && actual.ID_CHOFER) {
      enviarNotificacionAsignacion_(Object.assign({},resultado,{ID_CHOFER:actual.ID_CHOFER}), 'ASIGNACION_REASIGNADA');
    }
    if (nuevoChofer) {
      enviarNotificacionAsignacion_(resultado, 'ASIGNACION_REASIGNADA');
    }
    registrarEvento_({
      tipoEvento:'ASIGNACION_REASIGNADA',
      origen:'PANEL',
      idServicio:resultado.ID_SERVICIO,
      idAsignacion:resultado.ID_ASIGNACION,
      idVehiculo:resultado.ID_VEHICULO,
      idChofer:resultado.ID_CHOFER,
      descripcion:'Asignación reasignada; se procesó notificación push',
      datos:{choferAnterior:actual.ID_CHOFER,choferNuevo:resultado.ID_CHOFER,vehiculoAnterior:actual.ID_VEHICULO,vehiculoNuevo:resultado.ID_VEHICULO}
    });
  }

  return resultado;
}

function obtenerCuentaServicioFCM_() {
  const raw = PropertiesService.getScriptProperties().getProperty(FCM_CONFIG.SERVICE_ACCOUNT_PROPERTY);
  if (!raw) {
    throw new Error('Falta la propiedad de Script FCM_SERVICE_ACCOUNT_JSON. Pegá allí el JSON de la cuenta de servicio fcm-panel-choferes.');
  }
  let cred;
  try {
    cred = JSON.parse(raw);
  } catch (e) {
    throw new Error('FCM_SERVICE_ACCOUNT_JSON no contiene un JSON válido.');
  }
  if (!cred.client_email || !cred.private_key) {
    throw new Error('El JSON de la cuenta de servicio no contiene client_email/private_key.');
  }
  if (String(cred.project_id || '') !== FCM_CONFIG.PROJECT_ID) {
    throw new Error('La cuenta de servicio pertenece a otro proyecto: ' + String(cred.project_id || 'sin project_id'));
  }
  return cred;
}

function obtenerAccessTokenFCM_() {
  const cred = obtenerCuentaServicioFCM_();
  const ahora = Math.floor(Date.now()/1000);
  const header = {alg:'RS256',typ:'JWT'};
  const claims = {
    iss: cred.client_email,
    scope: 'https://www.googleapis.com/auth/firebase.messaging',
    aud: FCM_CONFIG.TOKEN_ENDPOINT,
    iat: ahora,
    exp: ahora + 3600
  };
  const encodedHeader = Utilities.base64EncodeWebSafe(JSON.stringify(header)).replace(/=+$/,'');
  const encodedClaims = Utilities.base64EncodeWebSafe(JSON.stringify(claims)).replace(/=+$/,'');
  const unsigned = encodedHeader + '.' + encodedClaims;
  const signature = Utilities.computeRsaSha256Signature(unsigned, cred.private_key);
  const encodedSignature = Utilities.base64EncodeWebSafe(signature).replace(/=+$/,'');
  const assertion = unsigned + '.' + encodedSignature;

  const response = UrlFetchApp.fetch(FCM_CONFIG.TOKEN_ENDPOINT, {
    method:'post',
    contentType:'application/x-www-form-urlencoded',
    payload:{
      grant_type:'urn:ietf:params:oauth:grant-type:jwt-bearer',
      assertion:assertion
    },
    muteHttpExceptions:true
  });
  const code = response.getResponseCode();
  const text = response.getContentText() || '';
  if (code < 200 || code >= 300) {
    throw new Error('OAuth FCM HTTP '+code+' — '+text);
  }
  let json;
  try { json = JSON.parse(text); } catch(e) { throw new Error('Respuesta OAuth FCM inválida.'); }
  if (!json.access_token) throw new Error('Google OAuth no devolvió access_token.');
  return json.access_token;
}

function enviarPushFCM_(token, titulo, cuerpo, datos, appUrl) {
  const accessToken = obtenerAccessTokenFCM_();
  const message = {
    token: String(token).trim(),
    notification: {
      title: String(titulo || 'Notificación'),
      body: String(cuerpo || '')
    },
    data: Object.keys(datos || {}).reduce(function(out,key){
      const v = datos[key];
      out[key] = v == null ? '' : String(v);
      return out;
    }, {})
  };

  const url = String(appUrl || '').trim();
  if (url) message.webpush = {fcm_options:{link:url}};

  const response = UrlFetchApp.fetch(FCM_CONFIG.ENDPOINT, {
    method:'post',
    contentType:'application/json',
    headers:{Authorization:'Bearer '+accessToken},
    payload:JSON.stringify({message:message}),
    muteHttpExceptions:true
  });

  const code = response.getResponseCode();
  const text = response.getContentText() || '';
  if (code < 200 || code >= 300) {
    let detalle = text;
    try {
      const j = JSON.parse(text);
      detalle = j.error?.status ? j.error.status+': '+(j.error.message || '') : text;
    } catch(e) {}
    throw new Error('FCM HTTP '+code+' — '+detalle);
  }
  try { return JSON.parse(text); } catch(e) { return {ok:true,raw:text}; }
}

function dvSyncNumero_(v) {
  if (v == null || String(v).trim() === '' || typeof v === 'boolean') return null;
  const n = Number(String(v).trim().replace(',', '.'));
  return Number.isFinite(n) ? n : null;
}

function dvSyncCampo_(r, nombres) {
  for (const nombre of nombres) {
    const key = Object.keys(r).find(k => normalizarClaveDV_(k) === normalizarClaveDV_(nombre));
    if (key && r[key] != null && String(r[key]).trim() !== '') return r[key];
  }
  return '';
}

function dvSyncHistorico_(rows) {
  return rows.map(r => ({

    cgm:dvSyncCampo_(r,['CGM']), zona:dvSyncCampo_(r,['Zona']),
    direccion:dvSyncCampo_(r,['Direccion','Dirección']),
    latitud:dvSyncNumero_(dvSyncCampo_(r,['Latitud'])), longitud:dvSyncNumero_(dvSyncCampo_(r,['Longitud'])),

    usuario:dvSyncCampo_(r,['Usuario']), nombre:dvSyncCampo_(r,['Nombre']),
    idRecorrido:dvSyncCampo_(r,['ID_RECORRIDO']), precision:dvSyncNumero_(dvSyncCampo_(r,['Precision'])),

function obtenerMapaDiaVerde_() {
  if (dvPanelRemotoActivo_()) {
    const remoto = dvPanelConsultarExclusivo_('mapaDiaVerde', arguments[0] || {}, false);
    return remoto;
  }

  const id = PropertiesService.getScriptProperties().getProperty('DIA_VERDE_HISTORICO_ID') || MAPA_DIA_VERDE.SPREADSHEET_ID;
  const ss = SpreadsheetApp.openById(id);
  const hojaGPS = ss.getSheetByName(MAPA_DIA_VERDE.HOJA_GPS);
  const hojaHistorico = ss.getSheetByName(MAPA_DIA_VERDE.HOJA_HISTORICO);
  if (!hojaGPS || !hojaHistorico) throw new Error('No se encontró GPS Día Verde o Hoja 1 en el histórico.');
  const gps = leerHojaMapaDV_(hojaGPS).map(r => ({
    timestamp:dvSyncCampo_(r,['Timestamp']), usuario:dvSyncCampo_(r,['Usuario']), nombre:dvSyncCampo_(r,['Nombre']),
    zona:dvSyncCampo_(r,['Zona']), cgm:dvSyncCampo_(r,['CGM']),
    latitud:dvSyncNumero_(dvSyncCampo_(r,['Latitud'])), longitud:dvSyncNumero_(dvSyncCampo_(r,['Longitud'])),
    precision:dvSyncNumero_(dvSyncCampo_(r,['Precision'])), rumbo:dvSyncNumero_(dvSyncCampo_(r,['Rumbo'])),
    activo:dvSyncCampo_(r,['Activo']), idPunto:dvSyncCampo_(r,['ID_PUNTO']),
    idRecorrido:dvSyncCampo_(r,['ID_RECORRIDO']), fuente:dvSyncCampo_(r,['FUENTE'])
  })).filter(p => p.latitud !== null && p.longitud !== null && Math.abs(p.latitud) <= 90 &&
    Math.abs(p.longitud) <= 180 && (p.latitud !== 0 || p.longitud !== 0));
  const historico = dvSyncHistorico_(leerHojaMapaDV_(hojaHistorico));
  const usuarios = {};
  gps.forEach(p => {
    const key = String(p.usuario || p.nombre).trim().toLowerCase();
    if (!usuarios[key]) usuarios[key] = {usuario:p.usuario, nombre:p.nombre, puntos:0};
    usuarios[key].puntos++;
  });
  const ordenados = gps.slice().sort((a,b) => new Date(a.timestamp) - new Date(b.timestamp));
  return {ok:true, syncVersion:2, fuente:'HISTORICO DIA VERDE',
    hojas:{gps:MAPA_DIA_VERDE.HOJA_GPS, historico:MAPA_DIA_VERDE.HOJA_HISTORICO},
    gps:gps, historico:historico, resumen:{puntosGPS:gps.length, direcciones:historico.length,
      nombres:Object.values(usuarios), ultimaPosicion:ordenados.length ? ordenados[ordenados.length-1] : null},
    timestamp:isoAhora_()};
}

function aplicarCorreccionesReclamosDiaVerde_(reclamos) {
  const correcciones = obtenerCorreccionesReclamosDiaVerde_();
  const porClave = {}, porDireccion = {};
  correcciones.forEach(c => {
    const lat = dvSyncNumero_(c.latitud), lng = dvSyncNumero_(c.longitud);
    if (lat === null || lng === null || Math.abs(lat) > 90 || Math.abs(lng) > 180 || (lat === 0 && lng === 0)) return;
    const dato = {latitud:lat, longitud:lng, usuario:c.usuario || '', fecha:c.fecha || ''};
    porClave[dvNormalizarClaveCorreccion_(c.clave)] = dato;
    const dir = dvSepararDireccionCorreccion_(c.direccion);
    if (c.localidad && c.zona && dir.calle) {
      porDireccion[dvClaveDireccionCorreccion_(c.localidad,c.zona,dir.calle,dir.numero)] = dato;
    }
  });
  return (Array.isArray(reclamos) ? reclamos : []).map(r => {
    const c = porClave[dvNormalizarClaveCorreccion_(r.clave)] ||
      porDireccion[dvClaveDireccionCorreccion_(r.localidad,r.zona,r.calle,r.numero)];
    // Una calle y altura de otro CGM nunca heredan esta ubicación.
    return c ? Object.assign({},r,{latitud:c.latitud,longitud:c.longitud,lat:c.latitud,lng:c.longitud,
      _corregidoPanel:true,_correccionUsuario:c.usuario,_correccionFecha:c.fecha}) : r;
  });
}

  }

  const ss = SpreadsheetApp.openById(RECLAMOS_DIA_VERDE.SPREADSHEET_ID);

function guardarCorreccionReclamo_(d) {
  if (dvPanelRemotoActivo_()) {
    const remoto = dvPanelConsultarExclusivo_('guardarCorreccionReclamo', arguments[0] || {}, true);
    dvPanelAuditarCorreccionRemota_('guardarCorreccionReclamo', remoto);
    return remoto;
  }

  const clave = String(d.clave || '').trim();
  const lat = dvSyncNumero_(d.latitud), lng = dvSyncNumero_(d.longitud);
  if (!clave || lat === null || lng === null || Math.abs(lat)>90 || Math.abs(lng)>180 || (lat===0 && lng===0)) {
    throw new Error('Clave y coordenadas válidas obligatorias.');
  }
  const lock = LockService.getScriptLock();
  lock.waitLock(30000);
  let resultado;
  try {
    const hoja = asegurarHojaCorreccionesDiaVerde_();
    const last = hoja.getLastRow();
    const claves = last>=2 ? hoja.getRange(2,1,last-1,1).getDisplayValues() : [];
    const indice = claves.findIndex(r=>dvNormalizarClaveCorreccion_(r[0])===dvNormalizarClaveCorreccion_(clave));
    const fecha = ahora_(), usuario = String(d.usuario || 'PANEL').trim();
    const valores = [[clave,lat,lng,String(d.direccion||''),String(d.localidad||''),String(d.zona||''),usuario,fecha]];
    hoja.getRange(indice<0 ? last+1 : indice+2,1,1,8).setValues(valores);
    SpreadsheetApp.flush();
    resultado = {clave:clave,latitud:lat,longitud:lng,fecha:fecha.toISOString(),usuario:usuario};
  } finally { lock.releaseLock(); }
  // La auditoría toma su propio lock; se ejecuta después de liberar el de la corrección.
  try { registrarEvento_({tipoEvento:'CORRECCION_PIN_DIA_VERDE',origen:'PANEL',
    descripcion:'Ubicación de reclamo compartida con Día Verde',datos:resultado}); }
  catch (error) { console.warn('Corrección guardada; auditoría pendiente.'); }
  return resultado;
}

function recDia_(valor){
  const s=String(valor||'').trim();
  if(/^\d{4}-\d{2}-\d{2}$/.test(s))return s;
  const m=s.match(/^(\d{2})\/(\d{2})\/(\d{4})$/);if(m)return m[3]+'-'+m[2]+'-'+m[1];
  return Utilities.formatDate(new Date(valor||ahora_()),CONFIG.ZONA_HORARIA,'yyyy-MM-dd');
}

function recActivo_(v,serviciosPorId){
  if(!['EN CURSO','ACTIVO','INICIADO'].includes(normalizar_(v.ESTADO)))return false;
  if(!v.ID_SERVICIO)return true;

  if(servicio?.FECHA && recDia_(servicio.FECHA)!==recDia_(v.HORA_INICIO||v.FECHA))return false;
  return !['FINALIZADO','CANCELADO','CANCELADA'].includes(normalizar_(servicio?.ESTADO));
}

// Reserva por chofer y vehículo. El bloqueo global solo protege la toma/liberación
// de estas claves; nunca las fotos en Drive. Caduca después del límite de ejecución.
function recTomarReservaViaje_(d){
  validarCampos_(d,['idChofer','idVehiculo']);
  const claves=['REC_ENVIO_CHOFER_'+String(d.idChofer).trim(),'REC_ENVIO_VEHICULO_'+String(d.idVehiculo).trim()];
  const token=Utilities.getUuid(),props=PropertiesService.getScriptProperties();
  const lock=LockService.getScriptLock();
  if(!lock.tryLock(1000))throw new Error('El backend está guardando otra operación. El envío se conserva pendiente para reintentar.');
  try{
    const ahora=Date.now();
    for(const clave of claves){
      const valor=props.getProperty(clave);let reserva=null;
      try{reserva=valor?JSON.parse(valor):null;}catch(e){}
      if(reserva?.vence>ahora)throw new Error('Ya se está procesando un envío de este chofer o vehículo. Se conserva pendiente para confirmar sin duplicarlo.');
    }
    const valor=JSON.stringify({token:token,vence:ahora+7*60*1000});
    const valores={};claves.forEach(k=>valores[k]=valor);props.setProperties(valores);
  }finally{lock.releaseLock();}
  return {releaseLock:function(){
    // No se usa un bloqueo global: una reserva nueva solo se toma tras liberar
    // ambas claves o vencer los siete minutos (la ejecución termina antes).
    for(const clave of claves){
      let reserva=null;try{reserva=JSON.parse(props.getProperty(clave)||'null');}catch(e){}
      if(reserva?.token===token)props.deleteProperty(clave);
    }
  }};
}

    }
    const conflictos=recViajesActivos_(viajes,v=>(recIgual_(v.ID_CHOFER,d.idChofer)||recIgual_(v.ID_VEHICULO,d.idVehiculo))&&!recIgual_(v.ID_VIAJE,existente?.ID_VIAJE));
    if(conflictos.some(v=>recIgual_(v.ID_CHOFER,d.idChofer)))throw new Error('Ya tenés un servicio en curso. Finalizalo antes de iniciar otro.');
    // Las filas históricas de otra jornada no reservan hoy la unidad de otro chofer.
    if(conflictos.some(v=>recIgual_(v.ID_VEHICULO,d.idVehiculo)&&recDia_(v.HORA_INICIO||v.FECHA)===recDia_(d.horaInicio||ahora_())))throw new Error('El vehículo tiene un servicio en curso de otro chofer en esta jornada. Revisá ese viaje desde el panel.');
    const rutas=recLeer_();let r=rutas.filter(r=>recValido_(r,d.idChofer,d.idVehiculo)).pop()||null;
    if(r&&!recFotos_(r))r=null;
    if(d.idRecorrido && (!r||!recIgual_(r.ID_RECORRIDO,d.idRecorrido)))throw new Error('La inspección ya no está vigente. Actualizá la pantalla y tomá cuatro fotos nuevas.');
    if(r){d.__recorridoId=r.ID_RECORRIDO;d.__inspeccionId=r.ID_INSPECCION;}
    else if(!existente && !['frente','trasera','lateral_derecho','lateral_izquierdo'].every(k=>!!d.fotos?.[k]))throw new Error('Se requieren las cuatro fotos para abrir un nuevo recorrido.');
    // Preparación aditiva de columnas; no cambia ni elimina filas anteriores.
    const ss=obtenerSpreadsheet_();
    if(!ss.getSheetByName(REC_CHOFER_HOJA))throw new Error('Falta preparar las inspecciones por recorrido en el backend.');
    let respuesta;
    if(secretaria&&existente){
      const fotos=registrarFotosSalida_(d,existente.ID_VIAJE);
      respuesta={ok:true,viaje:existente,FOTOS_SALIDA:fotos};
    }else respuesta=secretaria?iniciarViajeSecretariaBase_(d):iniciarViajeBase_(d);
    let viaje=respuesta.viaje||respuesta;const fotos=respuesta.FOTOS_SALIDA;
    if(!fotos||fotos.registradas!==4||fotos.errores.length||fotos.faltantes.length)return respuesta;
    let inspeccion=r?.ID_INSPECCION||viaje.ID_INSPECCION||respuesta.ID_INSPECCION;
    if(!inspeccion){inspeccion=generarID_('INS');insertarObjeto_(CONFIG.HOJAS.INSPECCIONES,{ID_INSPECCION:inspeccion,FECHA:viaje.HORA_INICIO,HORA:viaje.HORA_INICIO,ID_ASIGNACION:viaje.ID_ASIGNACION||'',ID_CHOFER:viaje.ID_CHOFER,ID_VEHICULO:viaje.ID_VEHICULO,KM_INICIAL:viaje.KM_INICIAL,RESULTADO:'APTO',ESTADO:'FINALIZADA',FECHA_FINALIZACION:ahora_(),OBSERVACIONES:'Inspección inicial del recorrido'});}
    if(!r){
      r=insertarObjeto_(REC_CHOFER_HOJA,{ID_RECORRIDO:generarID_('REC'),DIA:recDia_(viaje.HORA_INICIO),ID_CHOFER:d.idChofer,ID_VEHICULO:d.idVehiculo,ID_INSPECCION:inspeccion,ID_VIAJE_ORIGEN:viaje.ID_VIAJE,ESTADO:'ABIERTO',HORA_INICIO:viaje.HORA_INICIO,HORA_FIN:'',MOTIVO_CIERRE:''});
    }
    viaje=recVincular_(viaje,r,inspeccion);
    if(secretaria && existente && recActivo_(viaje)){
      const unidad=vehiculoSecretariaHabilitado_(viaje.ID_VEHICULO);
      const actual=normalizarKilometraje_(unidad.KM_ACTUAL),inicial=normalizarKilometraje_(viaje.KM_INICIAL);
      if(Number.isFinite(inicial)&&(!Number.isFinite(actual)||actual<inicial)){asegurarColumnaVehiculosKM_();actualizarPorID_(CONFIG.HOJAS.VEHICULOS,'ID_VEHICULO',viaje.ID_VEHICULO,{KM_ACTUAL:inicial});}
    }
    // Un cambio real de unidad invalida el recorrido anterior, incluso si luego vuelve a ella.
    for(const otra of rutas.filter(x=>(recIgual_(x.ID_CHOFER,d.idChofer)||recIgual_(x.ID_VEHICULO,d.idVehiculo))&&normalizar_(x.ESTADO)==='ABIERTO'&&!recIgual_(x.ID_RECORRIDO,r.ID_RECORRIDO)))recCerrarInterno_(otra.ID_RECORRIDO,'CAMBIO DE VEHICULO O JORNADA',null,viaje.ID_VIAJE);
    const confirmadas={...fotos,ID_RECORRIDO:r.ID_RECORRIDO,ID_INSPECCION:inspeccion,ID_VIAJE_ORIGEN:r.ID_VIAJE_ORIGEN};
    return secretaria?{...respuesta,viaje,FOTOS_SALIDA:confirmadas,RECORRIDO:r}:{...respuesta,...viaje,FOTOS_SALIDA:confirmadas,RECORRIDO:r};
  }finally{lock.releaseLock();}
}

function recVincularPanel_(viaje){
  if(viaje.ID_RECORRIDO)return viaje;
  const r=recLeer_().filter(r=>recValido_(r,viaje.ID_CHOFER,viaje.ID_VEHICULO)).pop();
  return r&&recFotos_(r)?recVincular_(viaje,r,r.ID_INSPECCION):viaje;
}

function repararIniciosServiciosHoy(){
  const hoy=recDia_(),viajes=obtenerViajes();let reparados=0;
  for(const servicio of obtenerServicios().filter(s=>recDia_(s.FECHA)===hoy&&normalizar_(s.ESTADO)==='EN CURSO'&&!s.HORA_INICIO)){
    const viaje=viajes.filter(v=>recIgual_(v.ID_SERVICIO,servicio.ID_SERVICIO)&&v.HORA_INICIO&&recDia_(v.HORA_INICIO)===hoy).sort((a,b)=>new Date(a.HORA_INICIO)-new Date(b.HORA_INICIO))[0];
    if(!viaje)continue;
    actualizarPorID_(CONFIG.HOJAS.SERVICIOS,'ID_SERVICIO',servicio.ID_SERVICIO,{HORA_INICIO:viaje.HORA_INICIO});
    if(!obtenerPorID_(CONFIG.HOJAS.SERVICIOS,'ID_SERVICIO',servicio.ID_SERVICIO)?.HORA_INICIO)throw new Error('No se pudo confirmar la reparación del inicio.');
    reparados++;
  }
  const faltantes=obtenerServicios().filter(s=>recDia_(s.FECHA)===hoy&&normalizar_(s.ESTADO)==='EN CURSO'&&!s.HORA_INICIO).length;
  const r={ok:true,reparados,serviciosEnCursoSinInicioHoy:faltantes};console.log(JSON.stringify(r));return r;
}
